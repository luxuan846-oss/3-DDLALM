\section{Convergence Analysis}
\begin{lemma}
Let $\{\bmu_k\}$ and $\{\lambda_k\}$ be the sequences generated by Algorithm \eqref{alg:peturbation linearized ADMM}. Then, we have
\begin{equation}
    \|\bmu_{k+1} - \bmu_k\|^2 \leq \frac{\delta_k^2}{4}, \qquad
    \|\lambda_k - \bmu_k\|^2 \leq \frac{\delta_k}{\tau_k},
\end{equation}
and
\begin{equation}
    \|\lambda_{k+1} - \lambda_k\|^2 \leq 2\|\lambda_{k+1} - \bmu_{k+1}\|^2 + 2\|\bmu_{k+1} - \lambda_k\|^2.
\end{equation}
\end{lemma}
