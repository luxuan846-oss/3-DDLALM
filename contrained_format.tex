\section{Introduction}
\label{sec:intro}
We consider the following nonconvex stochastic optimization problem with 
equality and inequality constraints:
\begin{align}\label{Prob: primal problem}
    \min_{\bx \in \calX} \quad & F(\bx) := f(\bx) + g(\bx), \notag \\
    \text{s.t.} \quad & \bc_{\calE}(\bx) = \bm{0}, \notag\\
    & \bc_{\calI}(\bx) \le \bm{0}, \tag{{\bf P}}
\end{align}
where $f(\bx) = \mathbb{E}_{\xi}[\mathbf{f}(\bx,\xi)]$ is a smooth nonconvex 
function with $\xi$ a random variable on a probability space $\Xi$ independent 
of $\bx$, and $g:\RR^n\to\RR\cup\{+\infty\}$ is a proper closed 
proximal-friendly function. The mappings $\bc_{\calE}:\RR^n\to\RR^{|\calE|}$ and
$\bc_{\calI}:\RR^n\to\RR^{|\calI|}$ are continuously differentiable. The set
$\calX\subseteq\RR^n$ is assumed to be nonempty and closed.
The feasible set is assumed to be nonempty and $F$ is lower 
bounded by $F^*>-\infty$. Exact evaluation of $\nabla f$ is often expensive or 
impractical, and thus we assume only access to stochastic gradients of $f$ at 
queried points.


\subsection{Related Work}
\label{sec:related-work}

\paragraph{Deterministic nonconvex constrained optimization.}
For deterministic nonconvex problems with equality and inequality constraints,
first-order methods with non-asymptotic complexity guarantees have been studied
extensively.
Proximal point methods~\cite{boob2023stochastic,ma2022quadratically} achieve
$\calO(\varepsilon^{-3})$ iteration complexity for smooth objectives via strongly
convex subproblems, and Boob et al.~\cite{boob2022level} further attain
$\calO(\varepsilon^{-2})$ by constructing quadratic models with adaptive
constraint levels.
Among penalty and augmented Lagrangian methods (ALMs), most approaches
require the penalty parameter to grow as $\Omega(\varepsilon^{-1})$, yielding
$\calO(\varepsilon^{-3})$ iteration complexity~\cite{li2021rate,jia2023first}.
Xie and Wright~\cite{xie2021proximal} achieve $\calO(\varepsilon^{-2})$
outer-loop complexity with a proximal ALM under a constant merit parameter,
but the algorithm is double-loop and the merit parameter must satisfy
$\rho = \Omega(\varepsilon^{-\eta})$ for some $\eta \in (0,1)$, making it
$\varepsilon$-dependent in theory.
SQP methods~\cite{curtis2021sqp,cartis2014complexity} also attain
$\calO(\varepsilon^{-2})$ complexity.

\paragraph{Stochastic nonconvex constrained optimization.}
We organize the literature by the two canonical settings:
\emph{semi-stochastic} (stochastic objective, deterministic constraints) and
\emph{fully stochastic} (both objective and constraints involve stochastic
oracles).

\emph{Semi-stochastic problems.}
Without additional structure on the noise, proximal point~\cite{boob2023stochastic,ma2022quadratically},
penalty~\cite{jin2022nested}, and SQP~\cite{curtis2021sqp} methods all achieve
$\calO(\varepsilon^{-4})$ oracle complexity in stochastic gradient evaluations,
matching the $\calO(\varepsilon^{-4})$ lower bound for unconstrained stochastic
optimization~\cite{arjevani2023lower}.
Under a mean-squared smoothness (MSS) condition,
variance reduction techniques---STORM~\cite{cutkosky2019storm} and
SPIDER~\cite{fang2018spider}---improve the complexity to $\calO(\varepsilon^{-3})$.
Representative results include the variance-reduced linearized ALM of
Shi et al.~\cite{shi2022linearized}, the truncated variance reduction approach
of Lu et al.~\cite{lu2022single}, and the inexact ALM of
Li et al.~\cite{li2021rate}, each attaining $\calO(\varepsilon^{-3})$ oracle
complexity under MSS for nonlinear constraints.

\emph{Fully stochastic problems.}
When constraint evaluations are also stochastic, complexity results are
uniformly higher.
Proximal point methods~\cite{boob2023stochastic,ma2022quadratically} incur
$\calO(\varepsilon^{-6})$ total oracle complexity due to the cost of solving
stochastic strongly convex subproblems.
Penalty-based ALMs employing STORM-type variance
reduction~\cite{alacaoglu2023augmented,cui2022complexity} face a large merit
parameter $\beta = \Theta(\varepsilon^{-1})$, which amplifies the variance of
the stochastic gradient and yields $\calO(\varepsilon^{-5})$ complexity even
under MSS.
Exact penalty methods combined with SPIDER~\cite{liu2022exact} achieve
$\calO(\varepsilon^{-4})$ (sub)gradient and $\calO(\varepsilon^{-6})$
constraint-evaluation complexity for nonsmooth problems.
Cui et al.~\cite{cui2024adaptive} recently propose an adaptive exact penalty
method for smooth \emph{equality}-constrained problems, obtaining
$\calO(\varepsilon^{-3})$ gradient and $\calO(\varepsilon^{-5})$
constraint-evaluation complexity; however, the analysis is confined to equality
constraints and relies on an exact penalty reformulation.
Stochastic SQP for equality-constrained fully stochastic problems~\cite{na2022sqp}
currently achieves only $\calO(\varepsilon^{-8})$.

\emph{Deterministic objective.}
The augmented Lagrangian (AL) framework, introduced by Hestenes~\cite{hestenes1969} and 
Powell~\cite{powell1969} and systematically developed by Bertsekas~\cite{bertsekas2014}, 
forms the algorithmic backbone for constrained optimization.
For nonconvex problems, early global convergence results were established for ADMM-type 
methods under boundedness assumptions on the dual iterates 
\cite{wang2019admm,bote2019,li2020global}.
From a complexity standpoint, Hajinezhad and Hong~\cite{hajinezhad2019} proposed a 
perturbed proximal primal-dual algorithm achieving $O(\varepsilon^{-4})$ iteration 
complexity, but requiring a feasible initialization.
Kong et al.~\cite{kong2019,kong2020} developed double-loop proximal AL methods attaining 
$O(\varepsilon^{-3})$ and further $O(\varepsilon^{-2.5})$ complexity under Slater's 
condition; however, these require solving subproblems in an inner loop, increasing 
implementation overhead.
The first single-loop method achieving $O(\varepsilon^{-2})$ iteration complexity was 
proposed by Zhang and Luo~\cite{zhang2020proximal,zhang2022global} via the SProx-ALM 
framework, exploiting Hoffman's error bound for the linear constraints.
More recently, Kim~\cite{kim2023} introduced the Proximal-Perturbed Lagrangian (PPAL) 
framework, which uses artificial variables and a dual smoothing term (``false penalty'') 
to enable a single-loop algorithm with $O(\varepsilon^{-2})$ complexity.
A distinctive feature of PPAL is that it works with a \emph{fixed} penalty parameter 
and does not require feasible initialization, Slater's condition, or full column rank of $A$.

\emph{Stochastic objective.}
When $f(x) = \mathbb{E}_\xi[F(x;\xi)]$ is an expectation function accessible only via 
stochastic oracles, the analysis becomes substantially more challenging due to gradient 
noise accumulation.
Shi, Wang, and Wang~\cite{shi2025mlalm} proposed MLALM, a single-loop linearized AL 
method incorporating a recursive momentum (STORM-type) technique 
for stochastic gradient estimation.
Under an unbounded increasing penalty schedule, MLALM achieves $O(\varepsilon^{-4})$ 
oracle complexity to find an $\varepsilon$-stationary point or $\varepsilon$-KKT point; 
this improves to $O(\varepsilon^{-3})$ only when the initial iterate is 
\emph{nearly feasible} within a prescribed accuracy.
Alacaoglu and Wright~\cite{alacaoglu2024} systematically studied single-loop variance-reduced 
AL and quadratic penalty algorithms across three constraint regimes:
for \emph{linear deterministic} constraints, they achieved $\widetilde{O}(\varepsilon^{-3})$ 
sample complexity using a constant penalty parameter and variance reduction, matching the 
lower bound for unconstrained stochastic nonconvex optimization;
for \emph{nonlinear deterministic} constraints, $\widetilde{O}(\varepsilon^{-4})$; 
and for \emph{stochastic nonlinear} constraints, $\widetilde{O}(\varepsilon^{-5})$.
Notably, they explicitly identified as an open problem the question of whether ALM with 
a \emph{fixed} penalty parameter (independent of $\varepsilon$) can achieve 
$O(\varepsilon^{-3})$ complexity for nonlinear equality constraints.
Lu, Mei, and Xiao~\cite{lu2026} addressed this gap partially via truncated recursive 
and Polyak momentum schemes, establishing $\widetilde{O}(\varepsilon^{-3})$ sample 
complexity together with \emph{high-probability} ($\varepsilon$-surely feasible) 
guarantees under an error bound condition; however, their analysis requires 
gradient clipping and relies on the error bound parameter $\theta \geq 1$, with 
the increasing penalty schedule remaining necessary.
In the PPAL setting with stochastic objectives, Zuo, Wang, and Wang~\cite{zuo2025adassp} 
proposed AdaSSP, an adaptive single-loop stochastic penalty method with high-probability 
complexity analysis; this method also employs an unbounded penalty sequence.

\paragraph{ALMs with fixed penalty parameter.}
A fundamental challenge in ALM design for nonlinear constraints is that
standard convergence analysis requires the penalty parameter to scale as
$\Omega(\varepsilon^{-1})$.
This growth inflates the Lipschitz constant of the augmented Lagrangian and
forces the step size to shrink proportionally, degrading the stochastic oracle
complexity.
For polyhedral (linear) constraints, the Luo--Pang global error
bound~\cite{luo1994error} decouples feasibility control from the penalty
magnitude.
Exploiting this structure, Zhang and Luo~\cite{zhang2022proximal} achieve
$\calO(\varepsilon^{-2})$ iteration complexity for deterministic nonconvex
problems under a genuinely constant penalty.
In the stochastic setting, Alacaoglu and
Wright~\cite{alacaoglu2024complexity} combine a fixed penalty with variance
reduction to attain the optimal $\calO(\varepsilon^{-3})$ sample complexity for
linear equality constraints; for nonlinear constraints, their single-loop
algorithm requires an increasing penalty and achieves only $\calO(\varepsilon^{-4})$.
Huang et al.~\cite{huang2025stochastic} extend the fixed-penalty result to
linear inequality constraints via Moreau envelope smoothing, also reaching
$\calO(\varepsilon^{-3})$.
For general nonlinear constraints, existing stochastic methods either require
an increasing penalty parameter~\cite{li2024stochastic,alacaoglu2024complexity},
yielding $\calO(\varepsilon^{-4})$ oracle complexity, or introduce dual
momentum to partially compensate, achieving
$\calO(\varepsilon^{-3.5})$~\cite{shi2025momentum}.
To date, no single-loop ALM with a genuinely constant penalty parameter has
been shown to achieve $\calO(\varepsilon^{-3})$ oracle complexity for nonlinear
nonconvex constraints in the stochastic setting.

The proposed Perturbed ALM addresses this gap.
By introducing a dual regularization term
$-\tfrac{\beta}{2}\|\blmbd - \bmu\|^2$ into the augmented Lagrangian, the
effective penalty parameter reduces to
$\rho = \alpha / (1 + \alpha\beta)$,
which is \emph{independent} of $\varepsilon$ for any fixed $\alpha, \beta > 0$.
This perturbation simultaneously enables a \emph{single-loop} primal--dual
update and achieves $\calO(\varepsilon^{-3})$ stochastic oracle complexity for
nonlinear nonconvex constrained optimization,
improving over the existing $\calO(\varepsilon^{-4})$ barrier for this setting.

\subsection{Stochastic KKT Point}
\begin{definition}[Stochastic KKT point]
\label{def:stationary}
A point $\bx^* \in \calX$ is called a \emph{stationary point} of
Problem~\eqref{Prob: primal problem} if there exists $\blmbd^* \in \RR^m$
such that the KKT conditions are satisfied:
\begin{equation}
    \mathbf{0} \in \nabla f(\bx^*) + \partial g(\bx^*)
    + J_{\bc}(\bx^*)^\top \blmbd^* + \mathcal{N}_{\calX}(\bx^*),
    \qquad \bc(\bx^*) = \mathbf{0}.
\end{equation}
\end{definition}

\subsection{Algorithm}

\textbf{Stochastic gradient estimator.}
Since the computation of exact gradient information of $f$ is costly and sometimes even prohibitive, we assume that a stochastic gradient at any inquiry point can be accessed. For a mini-batch sample $B_k$ at iteration $k$, let
\[
\nabla \mathbf{f}(\bx,B_k)
:=
\frac{1}{|B_k|}\sum_{\xi \in B_k} \nabla \mathbf{f}(\bx,\xi).
\]
Then, at the current iterate $\bx^k$, we compute the stochastic gradient estimator by
\begin{equation}\label{eq:storm}
    \bd^k
    =
    \nabla \mathbf{f}(\bx^k, B_k)
    + (1-\gamma_{k-1})
    \big(
        \bd^{k-1}
        - \nabla \mathbf{f}(\bx^{k-1}, B_k)
    \big),
    \qquad k \geq 1,
\end{equation}
with initialization
\begin{equation}\label{eq:storm-init}
    \bd^0 = \nabla \mathbf{f}(\bx^0,B_0).
\end{equation}
Here, the same mini-batch $B_k$ is used to evaluate both
$\nabla \mathbf{f}(\bx^k,B_k)$ and
$\nabla \mathbf{f}(\bx^{k-1},B_k)$.

We define \(\ \tilde{\nabla} \psi_{\rho}(\bx^k, \blmbd^k) := \bd^k + J_{\bc}(\bx^k)^\top\!\left(\blmbd^k + \rho\,\bc(\bx^k)\right)\ \) as the stochastic gradient estimator of $\nabla \varphi_{\rho}(\bx^k, \blmbd^k)$.    

\textbf{Update of $\bx$.}
\begin{equation}\label{eq:x-update}
\bx^{k+1} \in \argmin_{\bx \in \calX}
\left\{
% g(\bx)
% + \left\langle \bd^k + J_{\bc}(\bx^k)^\top\!\left(\blmbd^k + \rho\,\bc(\bx^k)\right),\ \bx - \bx^k \right\rangle
% + \frac{1}{2\eta_k}\|\bx - \bx^k\|^2
g(\bx) + \langle \tilde{\nabla} \psi_{\rho}(\bx^k, \blmbd^k),\ \bx - \bx^k \rangle + \frac{1}{2\eta_k}\|\bx - \bx^k\|^2
\right\}.
\end{equation}

\textbf{Update of $\bmu$.}
\begin{equation}\label{eq:mu-update}
    \bmu^{k+1} = \bmu^k + \theta_k\big(\blmbd^k - \bmu^k\big),
\end{equation} with a diminishing step-size
\begin{equation}\label{eq:theta-def}
    \theta_k = \min\!\left\{\theta_0,\; 
    \frac{\rho\delta_k}{\|\blmbd^k - \bmu^k\|^2 + 1}\right\},
\end{equation} and \(\delta_k \) satisfies
\begin{equation}\label{eq:delta-conditions}
    \delta_0 \in (0,1], \qquad 
    \lim_{k\to\infty} \delta_k = 0, \qquad \text{and} \qquad
    \sum_{k=0}^{\infty} \delta_k = +\infty.
\end{equation}

\textbf{Update of $\blmbd$.}
\begin{equation}\label{eq:lambda-update}
    \blmbd_{\calE}^{k+1} = \bmu_{\calE}^{k+1} + \rho\,\bc_{\calE}(\bx^{k+1}),
    \qquad
    \blmbd_{\calI}^{k+1} = \max\!\left(\bmu_{\calI}^{k+1} + \rho\,\bc_{\calI}(\bx^{k+1}),\;\bm{0}\right).
\end{equation}

\section{Stochastic Complexity Analysis}
\label{sec:analysis}
\begin{lemma}[Smoothness of $\varphi_{\rho}$]
\label{lem:phi-smooth}
Under Assumptions~\ref{Ass: boundedness of c and f} 
and~\ref{Ass: bounded multiplier}, the function 
$\varphi_{\rho}(\cdot,\blmbd)$ is $L_{\varphi}$-smooth for any 
$\blmbd$ with $\|\blmbd\|\leq B_{\lambda}$, i.e.,
\begin{equation}\label{eq:phi-smooth}
    \|\nabla_{\bx}\varphi_{\rho}(\bx,\blmbd)
     - \nabla_{\bx}\varphi_{\rho}(\by,\blmbd)\|
    \leq L_{\varphi}\|\bx-\by\|,
    \quad \forall\,\bx,\by\in\calX,
\end{equation}
where
\begin{equation}\label{eq:Lphi-def}
    L_{\varphi} := L_f + L_c(B_{\lambda}+\rho C) + \rho G^2.
\end{equation}
\end{lemma}

For simplicity, we introduce the gradient error estimator
\begin{align}
    \be^k
    &:= \nabla_{\bx}\varphi_{\rho}(\bx^k,\blmbd^k)
    - \tilde{\nabla}\psi_{\rho}(\bx^k,\blmbd^k) \notag\\
    &= \nabla f(\bx^k) - \bd^k.
\end{align}
Assume \(\gamma_k \equiv \gamma_f \in (0,1)\). It can be derived that, for \(k \ge 1\),
\[
\be^k = \sum_{t=0}^{k}(1-\gamma_f)^t\,\bu_{k-t},
\]
where
\begin{align}
\bu_k :=
\begin{cases}
\nabla \mathbf f(\bx^0,B_0) - \nabla f(\bx^0), & k=0,\\[2pt]
\nabla \mathbf f(\bx^k,B_k) - \nabla f(\bx^k) 
+(1-\gamma_f)\!\left(\nabla f(\bx^{k-1}) - \nabla \mathbf f(\bx^{k-1},B_k) \right) & k\ge1.
\end{cases}
\end{align}

\begin{lemma}
    Under Assumption~\ref{Ass: boundedness of c and f}-\ref{Ass: unbiased gradient oracle}. 
    Then for any \( K \geq 1\), it holds that with probability at least $1-2K \alpha$, that for any
    $k = 0, \cdots, K-1$, we have 
    \begin{align*}
    \|\be^k\|^2
\;\le\;
9\log(1/\alpha)\!\left(
\frac{a_f^2}{|B_0|}
+\frac{1}{\gamma_f}\max_{1\le j\le k}
\Big[
\frac{\gamma_f^2 a_f^2}{|B_j|}
+\frac{(1-\gamma_f)^2L_f^2\|\bx^j-\bx^{j-1}\|^2}{|B_j|}
\Big]
\right),
    \end{align*}
\end{lemma}
\begin{proof}
For $k=0$, by concentration for mini-batch average,
\[
\|\bu_0\|
\le 3a_f\sqrt{\frac{\log(1/\alpha)}{|B_0|}}
\quad\text{with probability at least }1-\alpha.
\]

For $k\ge 1$, decompose $\bu_k$ into a variance part and a drift part:
\[
\bu_k
=
\underbrace{
\frac{1}{|B_k|}\sum_{\xi\in B_k}
\!\left[
\gamma_f\big(\nabla f_\xi(\bx^k)-\nabla f(\bx^k)\big)
+(1-\gamma_f)\Big(
\nabla f_\xi(\bx^k)-\nabla f_\xi(\bx^{k-1})
-(\nabla f(\bx^k)-\nabla f(\bx^{k-1}))
\Big)
\right]}_{=: \text{sample fluctuation}}
+
(1-\gamma_f)\big(\nabla f(\bx^{k-1})-\nabla f(\bx^k)\big).
\]
Using concentration and smoothness,
\[
\|\bu_k\|
\le
3\gamma_f a_f\sqrt{\frac{\log(1/\alpha)}{|B_k|}}
+
6(1-\gamma_f)L_f\|\bx^k-\bx^{k-1}\|
\sqrt{\frac{\log(1/\alpha)}{|B_k|}}
\]
with probability at least $1-\alpha$.

Applying union bound over $k=0,\ldots,K-1$ gives an event of probability
at least $1-2K\alpha$ where all above bounds hold simultaneously.

On this event,
\[
\be^k=\sum_{t=0}^k(1-\gamma_f)^t\bu_{k-t}.
\]
By Cauchy-Schwarz (or weighted Jensen),
\[
\|\be^k\|^2
\le
\Big(\sum_{t=0}^k(1-\gamma_f)^t\Big)
\sum_{t=0}^k(1-\gamma_f)^t\|\bu_{k-t}\|^2
\le
\frac{1}{\gamma_f}\sum_{t=0}^k(1-\gamma_f)^t\|\bu_{k-t}\|^2.
\]
Substituting the high-probability bounds of $\bu_0,\bu_1,\ldots,\bu_k$
yields the claimed inequality.
\end{proof}

\begin{lemma}[Sufficient Descent]
\label{lem:sufficient-descent}
Let $\{(\bx^k, \blmbd^k, \bmu^k)\}$ be the sequence generated by
Algorithm~\ref{alg:PLADM-eq}. Under
Assumptions~\ref{Ass: boundedness of c and f},
\ref{Ass: unbiased gradient oracle}, and~\ref{Ass: bounded multiplier}, the following inequality holds for all $k \ge 0$:
\begin{align}
  \calL_{\rho}(\bx^{k+1}, \blmbd^{k+1}, \bmu^{k+1})
  &\;\le\;
  \calL_{\rho}(\bx^k, \blmbd^k, \bmu^k)
  - C_1^k\,\|\bx^{k+1}-\bx^k\|^2
  + \frac{\eta_k}{2}\|\be^k\|^2
  + \hat\delta_k,
  \label{eq:sufficient-descent}
\end{align}
where
\begin{equation}\label{eq:C1-deltahat-def}
  C_1^k := \frac{1}{2\eta_k} - \frac{L_\varphi}{2} - \rho L_c^2 > 0,
  \qquad
  \hat\delta_k := \frac{\rho\delta_k^2}{4} + \delta_k.
\end{equation}
Furthermore, rearranging \eqref{eq:sufficient-descent} gives
\begin{align}\label{eq:descent-rearranged}
  C_1^k\,\|\bx^{k+1}-\bx^k\|^2
  \leq\;
  &\calL_{\rho}(\bx^k, \blmbd^k, \bmu^k)
  - \calL_{\rho}(\bx^{k+1}, \blmbd^{k+1}, \bmu^{k+1})
  + \frac{\eta_k}{2}\|\be^k\|^2
  + \hat\delta_k.
\end{align}
\end{lemma}

\begin{lemma}[Violation Bound]\label{lem:feasibility}
Under the same conditions as Lemma~\ref{lem:sufficient-descent},
with $\delta_k = c/(k+1)$ for some $c > 0$, we have
\begin{equation}\label{eq:feasibility-rate}
  \textcolor{red}{%
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\left[\|\bc(\bx^{k+1})\|^2\right]
    = \mathcal{O}\!\left(\frac{\log T}{T}\right)
  }
\end{equation}
\end{lemma}

\begin{proof}
Let $r_k := \blmbd^k - \bmu^k$. By the $\blmbd$-update
\eqref{eq:lambda-update}, we have the identity
\begin{equation}\label{eq:r-feasibility}
  \|r_{k+1}\|^2 = \|\blmbd^{k+1} - \bmu^{k+1}\|^2 = \rho^2\|\bc(\bx^{k+1})\|^2,
\end{equation}
so it suffices to bound $\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}[\|r_{k+1}\|^2]$.

\medskip\noindent\textbf{Step 1: Derive the recursion for $\|r_{k+1}\|^2$.}
By the triangle inequality and $(a+b)^2 \leq 2(a^2+b^2)$,
\begin{equation}\label{eq:r-triangle}
  \|r_{k+1}\|^2
  \leq 2\|\blmbd^{k+1} - \blmbd^k\|^2 + 2\|\blmbd^k - \bmu^{k+1}\|^2.
\end{equation}
Applying Lemma~\ref{lem:basic-relations}~\eqref{eq:mu-lambda-diff}
to the second term and \eqref{eq:lambda-diff} to the first:
\begin{align}
  \|\blmbd^k - \bmu^{k+1}\|^2 &= (1-\theta_k)^2\|r_k\|^2, \notag\\
  \mathbb{E}[\|\blmbd^{k+1}-\blmbd^k\|^2 \mid \mathcal{F}_k]
  &\leq 2\rho^2 L_c^2\,\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2 \mid \mathcal{F}_k]
    + \frac{\rho^2\delta_k^2}{2}. \notag
\end{align}
Substituting into \eqref{eq:r-triangle} and using
$(1-\theta_k)^2 \leq (1-\theta_k)$ (since $\theta_k \in (0,1]$):
\begin{equation}\label{eq:r-recursion}
  \mathbb{E}[\|r_{k+1}\|^2 \mid \mathcal{F}_k]
  \leq 2(1-\theta_k)\|r_k\|^2
       + 4\rho^2 L_c^2\,\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2 \mid \mathcal{F}_k]
       + \rho^2\delta_k^2.
\end{equation}

\medskip\noindent\textbf{Step 2: Sum the recursion.}
Rearranging \eqref{eq:r-recursion}:
\begin{equation}
  2\theta_k\|r_k\|^2
  \leq 2\|r_k\|^2 - \mathbb{E}[\|r_{k+1}\|^2 \mid \mathcal{F}_k]
       + 4\rho^2 L_c^2\,\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2 \mid \mathcal{F}_k]
       + \rho^2\delta_k^2.
\end{equation}
Taking total expectation and summing over $k = 0,\ldots,T-1$:
\begin{equation}\label{eq:theta-r-sum}
  2\sum_{k=0}^{T-1}\theta_k\,\mathbb{E}[\|r_k\|^2]
  \leq 2\sum_{k=0}^{T-1}\mathbb{E}[\|r_k\|^2]
       - \sum_{k=0}^{T-1}\mathbb{E}[\|r_{k+1}\|^2]
       + 4\rho^2 L_c^2\sum_{k=0}^{T-1}\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2]
       + \rho^2\sum_{k=0}^{T-1}\delta_k^2.
\end{equation}
Note that the first two sums on the right telescope:
\begin{equation}\label{eq:telescope-r}
  2\sum_{k=0}^{T-1}\mathbb{E}[\|r_k\|^2] - \sum_{k=0}^{T-1}\mathbb{E}[\|r_{k+1}\|^2]
  = \sum_{k=0}^{T-1}\mathbb{E}[\|r_k\|^2] - \sum_{k=1}^{T-1}\mathbb{E}[\|r_k\|^2]
    + \mathbb{E}[\|r_0\|^2] - \mathbb{E}[\|r_T\|^2]
  \leq \mathbb{E}[\|r_0\|^2].
\end{equation}
Substituting \eqref{eq:telescope-r} into \eqref{eq:theta-r-sum}:
\begin{equation}\label{eq:theta-r-final}
  2\sum_{k=0}^{T-1}\theta_k\,\mathbb{E}[\|r_k\|^2]
  \leq \|r_0\|^2
       + 4\rho^2 L_c^2\sum_{k=0}^{T-1}\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2]
       + \rho^2\sum_{k=0}^{T-1}\delta_k^2.
\end{equation}
Dividing both sides of \eqref{eq:theta-r-final} by $T$:
  \textcolor{red}{
\begin{equation}\label{eq:feasibility-weighted}
  \frac{2}{T}\sum_{k=0}^{T-1}\theta_k\,\mathbb{E}[\|r_k\|^2]
  \leq \frac{\|r_0\|^2}{T}
       + \frac{4\rho^2 L_c^2}{T}\sum_{k=0}^{T-1}\mathbb{E}[\|\bx^{k+1}-\bx^k\|^2]
       + \frac{\rho^2}{T}\sum_{k=0}^{T-1}\delta_k^2.
\end{equation}}

\begin{lemma}[Vector Azuma-Hoeffding Inequality]
\label{lem:vector-azuma}
Let $\{\bv^k\}_{k=1}^K$ be a vector-valued martingale difference
sequence adapted to filtration $\{\calF_k\}_{k=0}^K$, satisfying
\begin{equation}
    \EE[\bv^k \mid \calF_{k-1}] = \zero,
    \qquad
    \|\bv^k\| \leq a_k,\quad k=1,\ldots,K.
\end{equation}
Then for any $\delta \in (0,1)$, with probability at least $1-\delta$,
\begin{equation}
    \left\|\sum_{k=1}^{K}\bv^k\right\|
    \leq
    3\sqrt{\log(1/\delta)\sum_{k=1}^{K}a_k^2}.
\end{equation}
\end{lemma}

\noindent
The above bound is a standard vector-valued Azuma-Hoeffding concentration result.
