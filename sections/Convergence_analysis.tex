\section{Auxiliary Lemmas}
\label{sec:basic-descent}
To analysis the convergence of the algorithms, 
This section establishes the basic descent properties of SPLALM.

For convenience, we introduce the successive differences of the iterates
\[
    \Delta_{\bx}^{k+1}:=\bx^{k+1}-\bx^k,
    \qquad
    \Delta_{\by}^{k+1}:=\by^{k+1}-\by^k,
    \qquad
    \Delta_{\bmu}^{k+1}:=\bmu^{k+1}-\bmu^k.
\]

Define
\[
    M_\mu:=\max\{\|\bmu^0\|,L_h\},
    \qquad
    C_r:=L_h+M_\mu,
\]
and introduce the splitting residual and the KKT-aligned multiplier
\[
\br^{k+1}:=\calA(\bx^{k+1})-\by^{k+1},
    \qquad
    \bz^{k+1}:=\bmu^k+\rho_{s_k}\br^{k+1}.
\]

Subtracting $\bmu^k$ from the $\bmu$-update and using the multiplier update
gives the identity
\begin{equation}
    \Delta_{\bmu}^{k+1}
    =
    \tau_k\br^{k+1},
    \qquad k\ge0.
    \label{eq:mu-drift-tau}
\end{equation}

\begin{lemma}
\label{lem:multiplier-boundedness}
\blue{Suppose that Assumption~\ref{ass:h} holds.}
Let $\{(\bx^k,\by^k,\bmu^k)\}_{k\ge0}$ be generated by
Algorithm~\ref{alg:ddlalm}.  Then, for
every $k\ge0$,
\begin{equation}
    \|\bmu^k\|\le M_\mu,
    \qquad
    \|\br^{k+1}\|
    \le
    \frac{C_r}{\rho_{s_k}},
    \qquad
    \|\Delta_{\bmu}^{k+1}\|
    \le
    \frac{\tau_k C_r}{\rho_{s_k}}.
    \label{eq:basic-bounds}
\end{equation}
Moreover, it holds that 
\begin{equation}
    \frac{1}{\tau_k}\|\Delta_{\bmu}^{k+1}\|^2
    % =\tau_k\|\br^{k+1}\|^2
    \le\frac{C_r^2\tau_k}{\rho_{s_k}^2}.
    \label{eq:dual-ascent-control}
\end{equation}
\end{lemma}

\begin{proof}
We establish the four estimates in turn. The exact $\by$-update gives the first-order
optimality condition
\[
    \zero\in\partial h(\by^{k+1})-\bmu^k-\rho_{s_k}\br^{k+1},
\]
hence $\bz^{k+1}:=\bmu^k+\rho_{s_k}\br^{k+1}\in\partial h(\by^{k+1})$.
By Assumption~\ref{ass:h}, it holds that
\begin{equation}
    \|\bz^{k+1}\|\le L_h,\qquad\forall\,k\ge0.
    \label{eq:anchor-bound}
\end{equation}

From $\bz^{k+1}=\bmu^k+\rho_{s_k}\br^{k+1}$ we have
$\br^{k+1}=(\bz^{k+1}-\bmu^k)/\rho_{s_k}$.  Substituting this into
\eqref{eq:mu-drift-tau} yields
\begin{equation}
    \bmu^{k+1}
    =\bmu^k+\tau_k\br^{k+1}
    =(1-q_k)\bmu^k+q_k\,\bz^{k+1},
    \qquad
    q_k:=\frac{\tau_k}{\rho_{s_k}}.
    \label{eq:mu-convex-combination}
\end{equation}
We verify $q_k\in(0,1)$.  Since $\tau_k>0$, we have $q_k>0$.
Since $\tau_k\le s_k$ and $\rho_{s_k}=\rho+s_k$ with $\rho>0$,
\[
    q_k=\frac{\tau_k}{\rho+s_k}\le\frac{s_k}{\rho+s_k}<1,
\]
where the strict inequality is due to the base penalty $\rho>0$, implying\eqref{eq:mu-convex-combination} is a convex combination.

We then prove $\|\bmu^k\|\le M_\mu$ for all $k\ge0$ by induction.

For the $k=0$ case,$\|\bmu^0\|\le M_\mu$ by definition of $M_\mu:=\max\{\|\bmu^0\|,L_h\}$.

For the $k \geq 1$ case, assume $\|\bmu^k\|\le M_\mu$.  Taking norms in
\eqref{eq:mu-convex-combination} and invoking \eqref{eq:anchor-bound} yield 
\[
    \|\bmu^{k+1}\|
    \le(1-q_k)\|\bmu^k\|+q_k\|\bz^{k+1}\|
    \le(1-q_k)M_\mu+q_k L_h
    \le M_\mu,
\]
where the last inequality uses $L_h\le M_\mu$ (which holds since
$M_\mu=\max\{\|\bmu^0\|,L_h\}\ge L_h$).  The induction is complete. By the triangle inequality, $\br^{k+1}=(\bz^{k+1}-\bmu^k)/\rho_{s_k}$, 
\eqref{eq:anchor-bound}, and $\|\bmu^k\|\le M_\mu$, we obtain 
\[
    \|\br^{k+1}\|
    \le\frac{\|\bz^{k+1}\|+\|\bmu^k\|}{\rho_{s_k}}
    \le\frac{L_h+M_\mu}{\rho_{s_k}}
    =\frac{C_r}{\rho_{s_k}},
\]
which proves the residual bound in \eqref{eq:basic-bounds}.  Because $\{s_k\}$ is nondecreasing, $\rho_{s_k}=\rho+s_k$ is nondecreasing, so
this bound is monotone nonincreasing in $k$.

Combining the residual bound with \eqref{eq:mu-drift-tau} gives the
increment estimate
\[
    \|\Delta_{\bmu}^{k+1}\|
    =\tau_k\|\br^{k+1}\|
    \le\frac{\tau_k C_r}{\rho_{s_k}}.
\]
Finally, substituting \eqref{eq:mu-drift-tau} and squaring,
\[
    \frac{1}{\tau_k}\|\Delta_{\bmu}^{k+1}\|^2
    =\tau_k\|\br^{k+1}\|^2
    \le\frac{C_r^2\tau_k}{\rho_{s_k}^2},
\]
which proves \eqref{eq:dual-ascent-control}, completing the proof.
\end{proof}

\blue{The next lemma first establishes the Lipschitz continuity of
$\nabla_{\bx}\Phi_s$ on a prescribed compact convex set for fixed
$s$, $\by$, and $\bmu$, and then gives a pointwise estimate at any
iteration whose current $\bx$-iterate lies in that set.}

\begin{lemma}
\label{lem:x-block-smoothness}
\blue{Suppose that Assumption~\ref{ass:h} holds and that
Assumption~\ref{ass:smoothness} holds on a given nonempty compact convex
set $\calS$.}
Then the following statements hold.
\begin{enumerate}[label={(\roman*)}]
\item Fix $\by,\bmu\in\RR^m$, and set $\rho_s:=\rho+s$. For all $\bx,\bx'\in\calS$,
\begin{equation}
\begin{aligned}
\left\|
\nabla_{\bx}\Phi_s(\bx,\by;\bmu)
-
\nabla_{\bx}\Phi_s(\bx',\by;\bmu)
\right\|
&\le L_x(s;\by,\bmu)\|\bx-\bx'\|,\\
L_x(s;\by,\bmu)
&=
L_f
+L_{\calA}\|\bmu-\rho_s\by\|
+\rho_s\bigl(L_{\calA}M_{\calA}+G_{\calA}^2\bigr).
\end{aligned}
\label{eq:static-x-block-smoothness}
\end{equation}
\item \blue{Let $\{(\bx^k,\by^k,\bmu^k)\}_{k\ge0}$ be generated by
Algorithm~\ref{alg:ddlalm}.  For any $k\ge0$ such that $\bx^k\in\calS$, it holds that}
\begin{equation}
    L_x(s_k;\by^k,\bmu^k)
    \le
    \widehat L\rho_{s_k},
    \label{eq:uniform-x-block-smoothness}
\end{equation}
where
\begin{equation}
    \widehat L
    :=
    \frac{L_f+L_{\calA}M_\mu}{\rho}
    +L_{\calA}(M_{\calA}+M_y)
    +G_{\calA}^2
    \label{eq:uniform-x-block-smoothness-constant}
\end{equation}
and
\[
    M_y
    :=
    \max\left\{
    \|\by^0\|,
    M_{\calA}+\frac{L_h+M_\mu}{\rho}
    \right\}.
\]
\blue{Here $L_f$, $L_{\calA}$, $M_{\calA}$, and $G_{\calA}$ are the
constants associated with Assumption~\ref{ass:smoothness} on $\calS$,
and $M_\mu$ is defined in Lemma~\ref{lem:multiplier-boundedness}; all
these constants are independent of $k$.}
\end{enumerate}
\end{lemma}

\begin{proof}
Fix $s\ge0$, $\by\in\RR^m$, and $\bmu\in\RR^m$, and set $\rho_s:=\rho+s$.
We prove items~{\rm (i)} and~{\rm (ii)} in turn.

\textbf{Proof of item~{\rm (i)}.}
By the definition of $\Phi_s$, the gradient of the $\bx$-block gives
\begin{equation}
    \nabla_{\bx}\Phi_s(\bx,\by;\bmu)
    =
    \nabla f(\bx)
    +J_{\calA}(\bx)^\top(\bmu-\rho_s\by)
    +\rho_s\,J_{\calA}(\bx)^\top\calA(\bx).
    \label{eq:grad-phi-x-block}
\end{equation}
We bound the Lipschitz constant of each term on $\calS$ separately.
By Assumption~\ref{ass:smoothness}\rm(i), $\nabla f$ is $L_f$-Lipschitz
on $\calS$.  Since $\bmu-\rho_s\by$ is a fixed vector and
$J_{\calA}$ is $L_{\calA}$-Lipschitz on $\calS$ by
Assumption~\ref{ass:smoothness}\rm(ii), the second term is
$L_{\calA}\|\bmu-\rho_s\by\|$-Lipschitz.  It remains to control the third
term, i.e., $J_{\calA}(\bx)^\top\calA(\bx)$.

For any $\bx,\bx'\in\calS$, we apply the identity
\[
    J_{\calA}(\bx)^\top\calA(\bx)
    -
    J_{\calA}(\bx')^\top\calA(\bx')
    =
    \bigl[J_{\calA}(\bx)-J_{\calA}(\bx')\bigr]^\top\calA(\bx)
    +
    J_{\calA}(\bx')^\top\bigl[\calA(\bx)-\calA(\bx')\bigr].
\]
Taking norms and using $\|\calA(\bx)\|\le M_{\calA}$ on $\calS$ and
$\|J_{\calA}(\bx')\|\le G_{\calA}$ by Assumption~\ref{ass:smoothness}\rm(ii),
we obtain
\[
    \bigl\|
    J_{\calA}(\bx)^\top\calA(\bx)
    -
    J_{\calA}(\bx')^\top\calA(\bx')
    \bigr\|
    \le
    L_{\calA}M_{\calA}\|\bx-\bx'\|
    +
    G_{\calA}\|\calA(\bx)-\calA(\bx')\|.
\]
Since $\calS$ is convex and $\|J_{\calA}(\bx)\|\le G_{\calA}$ on $\calS$, the
mean-value inequality gives
$\|\calA(\bx)-\calA(\bx')\|\le G_{\calA}\|\bx-\bx'\|$, and hence
\begin{equation}
    \bigl\|
    J_{\calA}(\bx)^\top\calA(\bx)
    -
    J_{\calA}(\bx')^\top\calA(\bx')
    \bigr\|
    \le
    \bigl(L_{\calA}M_{\calA}+G_{\calA}^2\bigr)\|\bx-\bx'\|.
    \label{eq:nonlinear-term-bound}
\end{equation}
Combining the Lipschitz constants of the three terms in
\eqref{eq:grad-phi-x-block} yields
\eqref{eq:static-x-block-smoothness}.

\textbf{Proof of item~{\rm (ii)}.}
{\color{blue}
Fix any $k\ge0$ such that $\bx^k\in\calS$.  If $k=0$, then
$\|\by^k\|=\|\by^0\|\le M_y$ by the definition of $M_y$.  If $k\ge1$,
Lemma~\ref{lem:multiplier-boundedness}, with the index shifted by one,
gives $\|\br^k\|\le C_r/\rho_{s_{k-1}}$, where $C_r=L_h+M_\mu$.
Since $\bx^k\in\calS$, we have $\|\calA(\bx^k)\|\le M_{\calA}$, and
therefore
\begin{equation}
\begin{aligned}
    \|\by^k\|
    &\le
    \|\calA(\bx^k)\|+\|\br^k\|\\
    &\le
    M_{\calA}+\frac{C_r}{\rho_{s_{k-1}}}
    \le
    M_{\calA}+\frac{L_h+M_\mu}{\rho}
    \le M_y.
\end{aligned}
\label{eq:y-iterate-bound}
\end{equation}
Thus $\|\by^k\|\le M_y$ in both cases.
}

We now apply item~{\rm (i)} with
$(s,\by,\bmu)=(s_k,\by^k,\bmu^k)$ and $\rho_s=\rho_{s_k}$.
Using $\|\bmu^k\|\le M_\mu$ and $\|\by^k\|\le M_y$,
\[
    \|\bmu^k-\rho_{s_k}\by^k\|
    \le
    \|\bmu^k\|+\rho_{s_k}\|\by^k\|
    \le
    M_\mu+\rho_{s_k} M_y.
\]
Substituting this bound into \eqref{eq:static-x-block-smoothness} yields
\[
    L_x(s_k;\by^k,\bmu^k)
    \le
    L_f+L_{\calA}M_\mu
    +\rho_{s_k}\bigl[L_{\calA}(M_{\calA}+M_y)+G_{\calA}^2\bigr].
\]
Since $\rho_{s_k}\ge\rho$, we have
$L_f+L_{\calA}M_\mu\le\frac{L_f+L_{\calA}M_\mu}{\rho}\,\rho_{s_k}$, and therefore
\[
    L_x(s_k;\by^k,\bmu^k)
    \le
    \rho_{s_k}
    \left[
    \frac{L_f+L_{\calA}M_\mu}{\rho}
    +L_{\calA}(M_{\calA}+M_y)
    +G_{\calA}^2
    \right]
    =
    \widehat L\,\rho_{s_k},
\]
\blue{Since the eligible index $k$ was arbitrary, this proves
\eqref{eq:uniform-x-block-smoothness}.}
\end{proof}

Let $\theta_x>0$ satisfy
$\theta_x\widehat L\le 1/2$, and choose
\begin{equation}
    \eta_k=\frac{\theta_x}{\rho_{s_k}},
    \qquad \forall k\ge0.
    \label{eq:explicit-primal-stepsize-coupling}
\end{equation}

\begin{lemma}
\label{lem:descent-skeleton}
\blue{Suppose that Assumption~\ref{ass:h} holds, that
$\{(\bx^k,\by^k,\bmu^k)\}_{k\ge0}$ is generated by
Algorithm~\ref{alg:ddlalm}, and that Assumption~\ref{ass:smoothness}
holds on a nonempty compact convex set $\calS$.}
\blue{Then, for every $k\ge0$ such that
$\bx^k,\bx^{k+1}\in\calS$, it holds that}
\begin{equation}\label{eq:variable-penalty-descent-multiplier}
\begin{aligned}
    &\Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    -\Phi_{s_k}(\bx^k,\by^k;\bmu^k)\\
    &\qquad\le
    -\frac{3\rho_{s_k}}{4\theta_x}\|\Delta_{\bx}^{k+1}\|^2
    -\frac{\rho_{s_k}}{2}\|\Delta_{\by}^{k+1}\|^2
    +\left(\frac{1}{\tau_k}+\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\tau_k^2}\right)
    \|\Delta_{\bmu}^{k+1}\|^2.
\end{aligned}
\end{equation}
\end{lemma}

\begin{proof}
We decompose the one-step change of the potential into a primal part
(from the $(\bx,\by)$-updates with $(\bmu^k,\rho_{s_k})$ held fixed) and
a dual part (from the $(\bmu,\rho_s)$-update with
$(\bx^{k+1},\by^{k+1})$ held fixed), and estimate them separately.
To lighten notation, write
\[
    L_k:=L_x(s_k;\by^k,\bmu^k),
\]
\blue{Since $\bx^k\in\calS$, the pointwise statement in
Lemma~\ref{lem:x-block-smoothness}\rm(ii) gives
$L_k\le\widehat L\,\rho_{s_k}$.}

\textbf{For the primal descent.}
Since $\calS$ is convex and $\bx^k,\bx^{k+1}\in\calS$, the segment
$[\bx^k,\bx^{k+1}]$ lies in $\calS$, so
Lemma~\ref{lem:x-block-smoothness} applies along it.
By the definition of $\Phi_s$ and the definition of $\bw^k$,
\[
    \nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
    =
    \nabla f(\bx^k)+J_{\calA}(\bx^k)^\top\bw^k.
\]
The $\bx$-update therefore gives
\[
    \bigl\langle\nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k),
              \Delta_{\bx}^{k+1}\bigr\rangle
    =
    -\frac{1}{\eta_k}\|\Delta_{\bx}^{k+1}\|^2.
\]
Applying the integral form of the descent lemma with the smoothness
constant $L_k$,
\begin{equation}
    \Phi_{s_k}(\bx^{k+1},\by^k;\bmu^k)
    -\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
    \le
    -\left(\frac{1}{\eta_k}-\frac{L_k}{2}\right)
    \|\Delta_{\bx}^{k+1}\|^2.
    \label{eq:x-block-descent}
\end{equation}
For the $\by$-block, $h$ is convex and the exact $\by$-update
the exact $\by$-update minimizes the $\rho_{s_k}$-strongly convex
subproblem $\Phi_{s_k}(\bx^{k+1},\cdot;\bmu^k)$; hence
\begin{equation}
    \Phi_{s_k}(\bx^{k+1},\by^{k+1};\bmu^k)
    -\Phi_{s_k}(\bx^{k+1},\by^k;\bmu^k)
    \le
    -\frac{\rho_{s_k}}{2}\|\Delta_{\by}^{k+1}\|^2.
    \label{eq:y-block-descent}
\end{equation}
Both \eqref{eq:x-block-descent} and \eqref{eq:y-block-descent} hold
$(\bmu^k,\rho_{s_k})$ fixed; adding them yields
\begin{equation}
    \Phi_{s_k}(\bx^{k+1},\by^{k+1};\bmu^k)
    -\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
    \le
    -\left(\frac{1}{\eta_k}-\frac{L_k}{2}\right)
    \|\Delta_{\bx}^{k+1}\|^2
    -\frac{\rho_{s_k}}{2}\|\Delta_{\by}^{k+1}\|^2.
    \label{eq:primal-descent}
\end{equation}

\textbf{For the dual ascent.}
Holding $(\bx^{k+1},\by^{k+1})$ fixed, the multiplier update and the
change of the effective penalty from $\rho_{s_k}$ to $\rho_{s_{k+1}}$
give
\[
\begin{aligned}
    &\Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    -\Phi_{s_k}(\bx^{k+1},\by^{k+1};\bmu^k)\\
    &\quad=
    \langle \Delta_{\bmu}^{k+1},\br^{k+1}\rangle
    +\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2}\|\br^{k+1}\|^2.
\end{aligned}
\]
By the identity $\Delta_{\bmu}^{k+1}=\tau_k\br^{k+1}$ from
\eqref{eq:mu-drift-tau} and the substitution
$\br^{k+1}=\Delta_{\bmu}^{k+1}/\tau_k$, it follows that
\begin{equation}
    \Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    -\Phi_{s_k}(\bx^{k+1},\by^{k+1};\bmu^k)
    =
    \left(\frac{1}{\tau_k}
    +\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\tau_k^2}\right)
    \|\Delta_{\bmu}^{k+1}\|^2.
    \label{eq:dual-ascent}
\end{equation}

\textbf{For the combination.}
Adding \eqref{eq:primal-descent} and \eqref{eq:dual-ascent} yields the
general one-step estimate
\begin{equation}
\begin{aligned}
    &\Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    -\Phi_{s_k}(\bx^k,\by^k;\bmu^k)\\
    &\qquad\le
    -\left(\frac{1}{\eta_k}-\frac{L_k}{2}\right)
    \|\Delta_{\bx}^{k+1}\|^2
    -\frac{\rho_{s_k}}{2}\|\Delta_{\by}^{k+1}\|^2
    +\left(\frac{1}{\tau_k}
    +\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\tau_k^2}\right)
    \|\Delta_{\bmu}^{k+1}\|^2.
\end{aligned}
\label{eq:variable-penalty-descent-general}
\end{equation}
Using $L_k\le\widehat L\,\rho_{s_k}$ and the stepsize coupling
$\eta_k=\theta_x/\rho_{s_k}$, the coefficient of
$\|\Delta_{\bx}^{k+1}\|^2$ is bounded below by
\[
    \frac{1}{\eta_k}-\frac{L_k}{2}
    \ge\frac{1}{\eta_k}-\frac{\widehat L\,\rho_{s_k}}{2}
    =\frac{\rho_{s_k}}{\theta_x}\!\left(1-\frac{\theta_x\widehat L}{2}\right)
    \ge\frac{3\rho_{s_k}}{4\theta_x},
\]
where the last inequality uses the assumed condition
$\theta_x\widehat L\le 1/2$.  Substituting this lower bound into
\eqref{eq:variable-penalty-descent-general} yields
\eqref{eq:variable-penalty-descent-multiplier}.
\end{proof}

\begin{lemma}[Potential upper bound and boundedness consequences]
\label{lem:bounded-iterates}
\blue{Suppose that Assumptions~\ref{ass:level-boundedness}
and~\ref{ass:h} hold, and let
$\{(\bx^k,\by^k,\bmu^k)\}_{k\ge0}$ be generated by
Algorithm~\ref{alg:ddlalm}.}  Assume that
\begin{equation}
\begin{aligned}
    C_e
    :=C_r^2\sum_{k=0}^{\infty}
    \left(
        \frac{\tau_k}{\rho_{s_k}^2}
        +\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\rho_{s_k}^2}
    \right)
    <\infty.
\end{aligned}
\label{eq:total-potential-error}
\end{equation}
Define
\begin{equation}
    \overline\Phi
    :=\Phi_{s_0}(\bx^0,\by^0;\bmu^0)+C_e,
    \qquad
    \alpha
    :=\overline\Phi+\frac{C_r^2}{2\rho_{s_0}},
    \label{eq:localized-level-value}
\end{equation}
and the level set
\begin{equation}
    \calS_\alpha:=\{\bx\in\RR^n:F(\bx)\le\alpha\}.
    \label{eq:derived-x-working-set}
\end{equation}
\blue{Suppose that Assumption~\ref{ass:smoothness} holds on a
nonempty compact convex set $\calS$ satisfying}
\begin{equation}
    \calS_\alpha\subseteq\operatorname{int}(\calS).
    \label{eq:level-set-interior-containment}
\end{equation}
\blue{Let $\widehat L$ be the constant defined in
Lemma~\ref{lem:x-block-smoothness}, and define}
\begin{equation}
    d_{\calS}:=\operatorname{dist}(\calS_\alpha,\partial\calS)>0,
    \qquad
    M_d:=\max\left\{
        \frac{\|\nabla_{\bx}\Phi_{s_0}(\bx^0,\by^0;\bmu^0)\|}{\rho_{s_0}},\,
        \frac{G_f+G_{\calA}(M_\mu+C_r)}{\rho_{s_0}}
    \right\},
    \label{eq:level-set-boundary-distance}
\end{equation}
Choose $\eta_k$ according to
\eqref{eq:explicit-primal-stepsize-coupling}, where $\theta_x>0$ satisfies
\begin{equation}
    \theta_x\widehat L\le\frac12,
    \qquad
    \theta_x M_d\le d_{\calS}.
    \label{eq:localized-primal-stepsize}
\end{equation}
Then \eqref{eq:variable-penalty-descent-multiplier} holds for every $k\ge0$,
and
\begin{equation}
    \Phi_{s_k}(\bx^k,\by^k;\bmu^k)\le\overline\Phi,
    \qquad
    F(\bx^k)\le\alpha.
    \label{eq:potential-and-objective-upper-bounds}
\end{equation}
Consequently, $\calS_\alpha$ is compact and
\begin{equation}
    \bx^k\in\calS_\alpha,
    \qquad k\ge0.
    \label{eq:derived-x-trajectory-bound}
\end{equation}
For later use, define
\begin{equation}
    C_{\calA}:=\sup_{\bx\in\calS_\alpha}\|\calA(\bx)\|,
    \qquad
    B_y:=\max\left\{\|\by^0\|,C_{\calA}+\frac{C_r}{\rho_{s_0}}\right\}.
    \label{eq:derived-operator-bounds}
\end{equation}
Then $C_{\calA}<\infty$, $\|\by^k\|\le B_y$ for every $k\ge0$, and the sequences
$\{\bx^k\}$, $\{\by^k\}$, and $\{\bmu^k\}$ are bounded.
\end{lemma}

\begin{proof}
Throughout the proof, write
\[
    e_k:=C_r^2\left(
        \frac{\tau_k}{\rho_{s_k}^2}
        +\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\rho_{s_k}^2}
    \right),
\]
so that $\sum_{k=0}^{\infty}e_k=C_e$ by \eqref{eq:total-potential-error}.

\textbf{For the potential-to-objective estimate.}
Since $h$ is $L_h$-Lipschitz and $\|\bmu^k\|\le M_\mu$ by
Lemma~\ref{lem:multiplier-boundedness},
\begin{equation}
\begin{aligned}
    F(\bx^k)
    &\le f(\bx^k)+h(\by^k)+L_h\|\br^k\|\\
    &=\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
      -\langle\bmu^k,\br^k\rangle
      -\frac{\rho_{s_k}}{2}\|\br^k\|^2
      +L_h\|\br^k\|\\
    &\le\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
      +C_r\|\br^k\|
      -\frac{\rho_{s_k}}{2}\|\br^k\|^2\\
    &\le\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
      +\frac{C_r^2}{2\rho_{s_k}},
\end{aligned}
\label{eq:potential-to-objective}
\end{equation}
where the last step completes the square and uses
$C_r=L_h+M_\mu$.

\textbf{For the inductive claim.}
\blue{The following is a bootstrap induction.  Starting from
$\bx^k\in\calS_\alpha$, we first establish $\bx^{k+1}\in\calS$ and only
then invoke the one-step descent estimate to prove
$\bx^{k+1}\in\calS_\alpha$.}
We prove jointly by induction on $k$ that
\begin{equation}
    \Phi_{s_k}(\bx^k,\by^k;\bmu^k)
    \le \Phi_{s_0}(\bx^0,\by^0;\bmu^0)
       +\sum_{j=0}^{k-1}e_j,
    \qquad
    \bx^k\in\calS_\alpha.
    \label{eq:boundedness-induction-claim}
\end{equation}

\textbf{For the base case.}
At $k=0$ the potential bound is immediate, and
\eqref{eq:potential-to-objective} gives
\[
    F(\bx^0)
    \le \Phi_{s_0}(\bx^0,\by^0;\bmu^0)
       +\frac{C_r^2}{2\rho_{s_0}}
    \le\alpha,
\]
so $\bx^0\in\calS_\alpha$ by the definition of $\calS_\alpha$.

\textbf{For the inductive step.}
Suppose \eqref{eq:boundedness-induction-claim} holds at index $k$.
We first show $\bx^{k+1}\in\calS$ and then close the induction.

\textbf{For the containment of $\bx^{k+1}$.}
By the $\bx$-update, the definition of $\bw^k$, and the definition of $\Phi_s$,
\[
    \Delta_{\bx}^{k+1}
    =
    -\eta_k \nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k).
\]
We claim $\|\nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k)\|/\rho_{s_k}\le M_d$ for every $k\ge0$.  For $k=0$ this
holds by the definition of $M_d$.  For $k\ge1$, using
$\|\bmu^k\|\le M_\mu$ from
Lemma~\ref{lem:multiplier-boundedness} and
$\|\br^k\|\le C_r/\rho_{s_{k-1}}$ from \eqref{eq:basic-bounds} (shifted
to index $k$), the normalization by $\rho_{s_k}$ gives
\[
    \frac{\|\bw^k\|}{\rho_{s_k}}
    \le\frac{M_\mu}{\rho_{s_k}}+\|\br^k\|
    \le\frac{M_\mu}{\rho_{s_0}}+\frac{C_r}{\rho_{s_0}},
\]
where the second inequality uses $\rho_{s_k}\ge\rho_{s_0}$ and
$\rho_{s_{k-1}}\ge\rho_{s_0}$.  Using
$\nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k)
=\nabla f(\bx^k)+J_{\calA}(\bx^k)^\top\bw^k$
together with $\|\nabla f(\bx^k)\|\le G_f$ and
$\|J_{\calA}(\bx^k)\|\le G_{\calA}$ from
Assumption~\ref{ass:smoothness} therefore gives
\[
    \frac{\|\nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k)\|}{\rho_{s_k}}
    \le\frac{G_f+G_{\calA}(M_\mu+C_r)}{\rho_{s_0}}
    \le M_d.
\]
In both cases,
\[
    \|\Delta_{\bx}^{k+1}\|
    =\frac{\theta_x}{\rho_{s_k}}\|\nabla_{\bx}\Phi_{s_k}(\bx^k,\by^k;\bmu^k)\|
    \le\theta_x M_d
    \le d_{\calS},
\]
where the last inequality uses \eqref{eq:localized-primal-stepsize}.
Since
$\bx^k\in\calS_\alpha$ and
$d_{\calS}=\operatorname{dist}(\calS_\alpha,\partial\calS)$
by \eqref{eq:level-set-boundary-distance}, this bound ensures
$\bx^{k+1}\in\calS$; convexity of $\calS$ then gives
$[\bx^k,\bx^{k+1}]\subseteq\calS$, so
Lemma~\ref{lem:descent-skeleton} applies at index $k$.

\textbf{For the potential update.}
Invoking \eqref{eq:variable-penalty-descent-multiplier} and using
$\|\Delta_{\bmu}^{k+1}\|\le\tau_k C_r/\rho_{s_k}$ from
\eqref{eq:basic-bounds} to bound the controlled-ascent term,
\[
    \Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    \le \Phi_{s_k}(\bx^k,\by^k;\bmu^k)+e_k
    \le \Phi_{s_0}(\bx^0,\by^0;\bmu^0)
       +\sum_{j=0}^{k}e_j,
\]
where the second inequality applies the inductive hypothesis.
Applying \eqref{eq:potential-to-objective} at index $k+1$ and using
$\sum_{j=0}^{k}e_j\le C_e$ together with $\rho_{s_{k+1}}\ge\rho_{s_0}$,
\[
    F(\bx^{k+1})
    \le\overline\Phi+\frac{C_r^2}{2\rho_{s_0}}
    =\alpha,
\]
so $\bx^{k+1}\in\calS_\alpha$, which closes the induction.

\textbf{For the boundedness consequences.}
The first part of \eqref{eq:boundedness-induction-claim} together with
$\sum_{j=0}^{k-1}e_j\le C_e$ proves
\eqref{eq:potential-and-objective-upper-bounds}.  By
Assumption~\ref{ass:level-boundedness}, $\calS_\alpha$ is compact, so
$C_{\calA}<\infty$ by continuity of $\calA$.  For $k\ge1$,
\eqref{eq:basic-bounds} gives
\[
    \|\by^k\|
    \le\|\calA(\bx^k)\|+\|\br^k\|
    \le C_{\calA}+\frac{C_r}{\rho_{s_{k-1}}}
    \le C_{\calA}+\frac{C_r}{\rho_{s_0}}
    \le B_y,
\]
and the case $k=0$ follows from the definition of $B_y$.  Finally,
$\|\bmu^k\|\le M_\mu$ by Lemma~\ref{lem:multiplier-boundedness},
so $\{\bx^k\}$, $\{\by^k\}$, and $\{\bmu^k\}$ are bounded.
\end{proof}

\begin{proposition}[Weighted finite length]
\label{lem:weighted-finite-length}
\blue{Suppose that the conditions of
Lemma~\ref{lem:bounded-iterates} hold.}
Assume in addition that
\[
    \rho_{s_k}=\rho+c_s(k+1)^a,
    \qquad
    c_s>0,\quad a\in(0,1).
\]
Assume also that the dual stepsizes satisfy
\begin{equation}
    \sum_{k=0}^{\infty}
    \frac{\tau_k}{\rho_{s_k}^2}
    <\infty.
    \label{eq:dual-ascent-summability}
\end{equation}
Then there exists a constant $C_{\mathrm{desc}}>0$, independent of $K$,
such that, for every $K\ge1$,
\begin{equation}\label{eq:weighted-finite-length}
    \sum_{k=0}^{K-1}
    \rho_{s_k}\left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right)
    \le
    C_{\mathrm{desc}} .
\end{equation}
\end{proposition}

\begin{proof}
Throughout the proof, $C>0$ denotes a generic constant independent of $k$
and $K$.  The value of $C$ may change from line to line and depends only on
the problem data and fixed algorithmic parameters.

\textbf{Step 1: Summability of the controlled ascent error.}
Define
\[
    e_k
    :=
    C_r^2\left(
    \frac{\tau_k}{\rho_{s_k}^2}
    +
    \frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\rho_{s_k}^2}
    \right).
\]
The first series in $e_k$ is finite by
\eqref{eq:dual-ascent-summability}.  For the penalty-drift series, the
concavity of $t\mapsto t^a$ and
$\rho_{s_k}=\rho+c_s(k+1)^a$ give
\[
    \rho_{s_{k+1}}-\rho_{s_k}
    =
    c_s\bigl((k+2)^a-(k+1)^a\bigr)
    \le
    c_sa(k+1)^{a-1}.
\]
Moreover, $\rho_{s_k}\ge c_s(k+1)^a$, so
\[
    \frac{\rho_{s_{k+1}}-\rho_{s_k}}{\rho_{s_k}^2}
    \le
    \frac{c_sa(k+1)^{a-1}}{c_s^2(k+1)^{2a}}
    =
    O(k^{-(1+a)}).
\]
Consequently, $\sum_{k=0}^{\infty}e_k<\infty$.
\blue{Lemma~\ref{lem:bounded-iterates} gives
$\bx^k\in\calS_\alpha$ for every $k\ge0$.}

\textbf{Step 2: Telescoping descent.}
In \eqref{eq:variable-penalty-descent-multiplier}, the ascent term
$\bigl(\frac{1}{\tau_k}+\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2\tau_k^2}\bigr)
\|\Delta_{\bmu}^{k+1}\|^2$ rewrites as
$\tau_k\|\br^{k+1}\|^2+\frac{\rho_{s_{k+1}}-\rho_{s_k}}{2}\|\br^{k+1}\|^2$
via $\Delta_{\bmu}^{k+1}=\tau_k\br^{k+1}$, and is bounded by $e_k$ using
\eqref{eq:basic-bounds}.
By \eqref{eq:variable-penalty-descent-multiplier}, \blue{set
$c_{\mathrm{desc}}:=\min\{3/(4\theta_x),1/2\}>0$.  Then}
\begin{equation}\label{eq:one-step-descent-clean}
    \Phi_{s_{k+1}}(\bx^{k+1},\by^{k+1};\bmu^{k+1})
    -
    \Phi_{s_k}(\bx^k,\by^k;\bmu^k)
    \le
    -c_{\mathrm{desc}}\rho_{s_k}
    \left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right)
    +e_k,
\end{equation}
Summing \eqref{eq:one-step-descent-clean} from $k=0$ to $K-1$ yields
\[
    c_{\mathrm{desc}}
    \sum_{k=0}^{K-1}
    \rho_{s_k}\left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right)
    \le
    \Phi_{s_0}(\bx^0,\by^0;\bmu^0)
    -
    \Phi_{s_K}(\bx^K,\by^K;\bmu^K)
    +
    \sum_{k=0}^{K-1}e_k.
\]

\textbf{Step 3: Lower bound for the potential.}
{\color{blue}
Since $h$ is globally $L_h$-Lipschitz and
$\br^K=\calA(\bx^K)-\by^K$,
\[
    h(\by^K)
    \ge h\bigl(\calA(\bx^K)\bigr)-L_h\|\br^K\|.
\]
Using this inequality, $\|\bmu^K\|\le M_\mu$, and completing the square give
\[
\begin{aligned}
    \Phi_{s_K}(\bx^K,\by^K;\bmu^K)
    &=
    f(\bx^K)+h(\by^K)
    +\langle\bmu^K,\br^K\rangle
    +\frac{\rho_{s_K}}{2}\|\br^K\|^2\\
    &\ge
    F(\bx^K)-C_r\|\br^K\|
    +\frac{\rho_{s_K}}{2}\|\br^K\|^2\\
    &\ge
    \inf_{\bx\in\calS_\alpha}F(\bx)
    -\frac{C_r^2}{2\rho_{s_K}}\\
    &\ge
    \inf_{\bx\in\calS_\alpha}F(\bx)
    -\frac{C_r^2}{2\rho_{s_0}}.
\end{aligned}
\]
The infimum is finite because $F$ is continuous on the compact set
$\calS_\alpha$.
Therefore,
\[
    \sum_{k=0}^{K-1}
    \rho_{s_k}\left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right)
    \le
    \frac{1}{c_{\mathrm{desc}}}
    \left[
    \Phi_{s_0}(\bx^0,\by^0;\bmu^0)
    -
    \inf_{\bx\in\calS_\alpha}F(\bx)
    +
    \frac{C_r^2}{2\rho_{s_0}}
    +
    \sum_{k=0}^{\infty}e_k
    \right]
    =:
    C_{\mathrm{desc}},
\]
where $C_{\mathrm{desc}}$ is finite and independent of $K$.
}
\end{proof}

\begin{remark}[Dual stepsize design]
\label{rmk:dual-stepsize-choice}
Lemma~\ref{lem:multiplier-boundedness} shows that multiplier boundedness only
requires $0<\tau_k\le s_k$.  \blue{The additional condition needed for the
descent argument is}
\[
    \sum_{k=0}^{\infty}\frac{\tau_k}{\rho_{s_k}^2}
    <\infty.
\]
For $s_k=c_s(k+1)^a$, a family satisfying
$\tau_k=O\bigl(s_k/(k+1)^q\bigr)$ meets this requirement whenever
$q>1-a$.  If a downstream result additionally requires finite total variation
of the anchor multiplier, Lemma~\ref{lem:multiplier-boundedness} shows that the
stronger condition $\sum_k\tau_k/\rho_{s_k}<\infty$ is sufficient.  The global
complexity result below uses $q=1$, namely
$\tau_k=\theta_\mu s_k/(k+2)$.  This choice makes the potential-ascent error
summable and gives
$\operatorname{dist}(\bw^k,\partial h(\by^k))=O(k^{-1})$; finite total
variation of $\{\bmu^k\}$ is not needed for that result.
\end{remark}
\begin{theorem}[Global KKT residual complexity]
\label{theorem:target-complexity}
{\color{blue}
Suppose that Assumptions~\ref{ass:level-boundedness} and~\ref{ass:h} hold.
Let $c_s>0$, $a\in(0,1)$, $\theta_x>0$, and
$\theta_\mu\in(0,1]$, and choose
\begin{equation}
\label{eq:complexity-parameter-schedule}
    s_k=c_s(k+1)^a,
    \qquad
    \rho_{s_k}=\rho+s_k,
    \qquad
    \eta_k=\frac{\theta_x}{\rho_{s_k}},
    \qquad
    \tau_k=\frac{\theta_\mu s_k}{k+2},
\end{equation}
Let $C_e$, $\alpha$, and $\calS_\alpha$ be defined by
\eqref{eq:total-potential-error}--\eqref{eq:derived-x-working-set}.  Suppose
that Assumption~\ref{ass:smoothness} holds on a nonempty compact convex set
$\calS$ satisfying
$\calS_\alpha\subseteq\operatorname{int}(\calS)$.  Let $M_d$ and
$\widehat L$ be defined as in
Lemma~\ref{lem:bounded-iterates}, define
$d_{\calS}$ by \eqref{eq:level-set-boundary-distance}, and require
\begin{equation}
    \theta_x\widehat L\le\frac12,
    \qquad
    \theta_x M_d\le d_{\calS}.
    \label{eq:complexity-localized-primal-stepsize}
\end{equation*}
}

For any $K\ge2$, choose
\[
    \kappa
    \in
    \argmin_{k\in\{\lceil K/2\rceil,\ldots,K-1\}}
    \rho_{s_k}\left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right).
\]
Define the effective KKT multiplier
\[
    \bz^\kappa
    :=
    \bmu^{\kappa-1}+\rho_{s_{\kappa-1}}\br^\kappa,
\]
where $\br^\kappa:=\calA(\bx^\kappa)-\by^\kappa$.
Then
\begin{equation}\label{eq:kkt-residual-general-a}
\begin{aligned}
    \left\|
        \nabla f(\bx^\kappa)
        +J_{\calA}(\bx^\kappa)^\top \bz^\kappa
    \right\|
    &=O\!\left(K^{-(1-a)/2}\right),\\
    \operatorname{dist}\bigl(\bz^\kappa,\partial h(\by^\kappa)\bigr)
    &=0,\\
    \|\br^\kappa\|
    &=O(K^{-a}).
\end{aligned}
\end{equation}
Moreover, $\|\bz^\kappa\|\le L_h$.  In particular, for the balanced
choice $a=1/3$,
\[
    \max\left\{
        \left\|
            \nabla f(\bx^\kappa)
            +J_{\calA}(\bx^\kappa)^\top \bz^\kappa
        \right\|,
        \operatorname{dist}\bigl(
            \bz^\kappa,\partial h(\by^\kappa)
        \bigr),
        \|\br^\kappa\|
    \right\}
    =O(K^{-1/3}).
\]
Consequently, to obtain an $\epsilon$-approximate KKT point in the sense that
\[
    \left\|
    \nabla f(\bx^\kappa)
    +
    J_{\calA}(\bx^\kappa)^\top \bz^\kappa
    \right\|
    \le
    \epsilon,
    \qquad
    \operatorname{dist}\bigl(
        \bz^\kappa,\partial h(\by^\kappa)
    \bigr)
    \le
    \epsilon,
    \qquad
    \|\br^\kappa\|
    \le
    \epsilon,
\]
under the assumptions above, it suffices to take
\[
    K
    =
    O(\epsilon^{-3}).
\]
Moreover, the virtual multiplier
$\blmbd^\kappa=\bmu^{\kappa-1}+s_{\kappa-1}\br^\kappa$
and the KKT-aligned multiplier
$\bz^\kappa=\bmu^{\kappa-1}+\rho_{s_{\kappa-1}}\br^\kappa$
satisfy
\[
    \bz^\kappa-\blmbd^\kappa
    =
    \rho\br^\kappa,
\]
and hence
\[
    \operatorname{dist}\bigl(
        \blmbd^\kappa,\partial h(\by^\kappa)
    \bigr)
    \le \rho\|\br^\kappa\|
    =O(K^{-a}).
\]
The multiplier driving the $\bx$-step also satisfies
\[
    \operatorname{dist}\bigl(
        \bw^\kappa,\partial h(\by^\kappa)
    \bigr)
    =O(K^{-1}).
\]
The KKT residual is stated with $\bz^\kappa$ because
$\bz^\kappa\in\partial h(\by^\kappa)$ by the preceding $\by$-step
optimality condition.
\end{theorem}

\begin{proof}
The schedule \eqref{eq:complexity-parameter-schedule} satisfies
$0<\tau_k\le s_k$, and
\[
\begin{aligned}
    \sum_{k=0}^{\infty}\frac{\tau_k}{\rho_{s_k}^2}
    &\le
    \theta_\mu
    \sum_{k=0}^{\infty}
    \frac{1}{(k+2)\rho_{s_k}}\\
    &\le
    \frac{\theta_\mu}{c_s}
    \sum_{k=0}^{\infty}
    \frac{1}{(k+2)(k+1)^a}
    <\infty.
\end{aligned}
\]
The same schedule satisfies
\[
    \frac{\rho_{s_{k+1}}-\rho_{s_k}}{\rho_{s_k}^2}
    =O(k^{-(1+a)}),
\]
\blue{so \eqref{eq:total-potential-error} holds.  Therefore, the
assumptions on $\calS$ and
\eqref{eq:complexity-localized-primal-stepsize} verify all conditions of
Lemma~\ref{lem:bounded-iterates}.  The bootstrap induction in that lemma
establishes, rather than assumes, that
$\bx^k\in\calS_\alpha\subseteq\calS$ for every $k\ge0$.  Hence all
conditions of Proposition~\ref{lem:weighted-finite-length} hold, and
that proposition gives}
\[
    \sum_{k=0}^{K-1}
    \rho_{s_k}\left(\|\Delta_{\bx}^{k+1}\|^2+\|\Delta_{\by}^{k+1}\|^2\right)
    \le
    C_{\mathrm{desc}},
    \qquad
    \forall K\ge1.
\]
The index set $\{\lceil K/2\rceil,\ldots,K-1\}$ has
$\lfloor K/2\rfloor$ elements, and $\lfloor K/2\rfloor\ge K/3$ for every
$K\ge2$.  Hence, by the definition of $\kappa$,
\[
    \rho_{s_\kappa}\left(\|\Delta_{\bx}^{\kappa+1}\|^2+\|\Delta_{\by}^{\kappa+1}\|^2\right)
    \le
    \frac{3C_{\mathrm{desc}}}{K}.
\]
Since $\kappa\in\{\lceil K/2\rceil,\ldots,K-1\}$ and
$\rho_{s_k}=\rho+c_s(k+1)^a$, we have $\rho_{s_\kappa}\asymp K^a$.  Thus
\[
    \|\Delta_{\bx}^{\kappa+1}\|+\|\Delta_{\by}^{\kappa+1}\|
    =
    O\!\left(K^{-(1+a)/2}\right).
\]

\textbf{Step 1: Feasibility and multiplier residuals.}
By Lemma~\ref{lem:multiplier-boundedness},
\[
    \|\br^\kappa\|
    \le
    \frac{C_r}{\rho_{s_{\kappa-1}}}
    =
    O(K^{-a}).
\]
The exact $\by$-update optimality condition and
\eqref{eq:bounded-h-subgradients} give
\[
    \bz^\kappa\in\partial h(\by^\kappa),
    \qquad
    \|\bz^\kappa\|\le L_h,
\]
which proves the second residual in
\eqref{eq:kkt-residual-general-a}.  Moreover,
$\bz^\kappa-\blmbd^\kappa=\rho\br^\kappa$ gives the stated virtual-multiplier
residual.

For every $k\ge1$, the definitions of $\bw^k$ and $\bz^k$, together with
\eqref{eq:mu-drift-tau}, give
\[
\begin{aligned}
    \bw^k-\bz^k
    &=
    \bmu^k-\bmu^{k-1}
    +\bigl(\rho_{s_k}-\rho_{s_{k-1}}\bigr)\br^k\\
    &=
    \bigl(
        \tau_{k-1}+\rho_{s_k}-\rho_{s_{k-1}}
    \bigr)\br^k.
\end{aligned}
\]
Using \eqref{eq:basic-bounds} with index $k-1$ and the schedule
\eqref{eq:complexity-parameter-schedule}, we obtain
\[
\begin{aligned}
    \|\bw^k-\bz^k\|
    &\le
    C_r\left(
        \frac{\tau_{k-1}}{\rho_{s_{k-1}}}
        +
        \frac{\rho_{s_k}-\rho_{s_{k-1}}}
             {\rho_{s_{k-1}}}
    \right)\\
    &\le
    C_r\left(
        \frac{\theta_\mu}{k+1}
        +\frac{a}{k}
    \right)
    =O(k^{-1}),
\end{aligned}
\]
where the second inequality uses
$(k+1)^a-k^a\le ak^{a-1}$.  Since the exact $\by$-update optimality
condition gives $\bz^k\in\partial h(\by^k)$,
\[
    \operatorname{dist}\bigl(\bw^k,\partial h(\by^k)\bigr)
    \le \|\bw^k-\bz^k\|
    =O(k^{-1}).
\]
In particular, $\kappa\ge K/2$ gives
$\|\bw^\kappa-\bz^\kappa\|=O(K^{-1})$.

\textbf{Step 2: Stationarity residual.}
The $\bx$-update optimality condition gives
\[
    \nabla f(\bx^\kappa)
    +
    J_{\calA}(\bx^\kappa)^\top \bw^\kappa
    =
    -\frac{1}{\eta_\kappa}\Delta_{\bx}^{\kappa+1},
\]
where $\bw^\kappa:=\bmu^\kappa+\rho_{s_\kappa}\br^\kappa$.
Consequently,
\[
\begin{aligned}
    \left\|
        \nabla f(\bx^\kappa)
        +J_{\calA}(\bx^\kappa)^\top \bz^\kappa
    \right\|
    &\le
    \frac{1}{\eta_\kappa}\|\Delta_{\bx}^{\kappa+1}\|
    +G_{\calA}\|\bw^\kappa-\bz^\kappa\|\\
    &=O\!\left(K^{-(1-a)/2}\right)+O(K^{-1})\\
    &=O\!\left(K^{-(1-a)/2}\right).
\end{aligned}
\]

The two residual rates are balanced by solving
$a=(1-a)/2$, which gives $a=1/3$. For this choice, both stationarity and
splitting residuals are $O(K^{-1/3})$, and the iteration complexity claim follows by
setting $K=O(\epsilon^{-3})$.
\end{proof}

\begin{remark}
\label{rmk:first-order-eps-minus-three}
The schedule \eqref{eq:complexity-parameter-schedule} couples the dual
regularization, primal step, and dual step as
\[
    \frac{1}{s_k}=\Theta(k^{-a}),
    \qquad
    \eta_k=\Theta(k^{-a}),
    \qquad
    \tau_k=\Theta(k^{a-1}).
\]
The dual-ascent and penalty-drift errors then obey
\[
    \frac{\tau_k}{\rho_{s_k}^2}
    =O(k^{-(1+a)}),
    \qquad
    \frac{\rho_{s_{k+1}}-\rho_{s_k}}{\rho_{s_k}^2}
    =O(k^{-(1+a)}),
\]
so both error sequences are summable.  In addition,
\[
    \|\Delta_{\bmu}^{k+1}\|=O(k^{-1}),
    \qquad
    \operatorname{dist}\bigl(\bw^k,\partial h(\by^k)\bigr)=O(k^{-1}).
\]
Thus the splitting residual is retained as the feasibility component of the
KKT residual, while its contribution to the potential ascent is converted to
the summable multiplier-increment term
\[
    \tau_k\|\br^{k+1}\|^2
    =
    \frac{1}{\tau_k}\|\Delta_{\bmu}^{k+1}\|^2.
\]
Finally, weighted finite length gives
\[
    r_{\rm stat}=O(K^{-(1-a)/2}),
    \qquad
    r_{\rm feas}=O(K^{-a}).
\]
Balancing these rates yields $a=1/3$.  Hence
\[
    \frac{1}{s_k}=\Theta(k^{-1/3}),
    \qquad
    \eta_k=\Theta(k^{-1/3}),
    \qquad
    \tau_k=\Theta(k^{-2/3}),
\]
and all three KKT residuals are $O(K^{-1/3})$.
\end{remark}

\section{Complexity Analysis}
\label{sec:complexity-analysis}
\label{sec:basic-descent}
Let us define the constants
\begin{equation}
    L_{\calD,\bx}
    :=
    L_f + L_{\calA}L_h + \rho G_{\calA}^{2},
    \label{eq:x-weak-convexity-modulus}
\end{equation}
and 
\[
    \mu_\gamma
    :=
    \frac{1}{\gamma}-L_{\calD,\bx},
    \qquad
    L_x
    :=
    L_{\calD,\bx}+\frac{1}{\gamma},
    \qquad
    L_\lambda(s)
    :=
    \frac{1}{\rho}+\frac{1}{s},
\] where $L_f$, $L_{\calA}$, $L_h$, $\rho$, and $G_{\calA}$ defined as before. 

The following proposition summarizes the curvature and smoothness properties
of the smoothed perturbed potential function $\mathcal P_s$.
The proof is deferred to Appendix~\ref{app:curvature-smoothness}.

\begin{proposition}
\label{prop:curvature-smoothness}
Suppose that Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold, and let the smoothing parameter  
$ 0 < \gamma < \frac{1}{2L_{\calD,\bx}}$. Then, for all $s>0$, $\mathcal P_s$ is
$\mu_\gamma $-strongly convex in $\bx$ and
$1/s$-strongly concave in $\blmbd$. Moreover, for all
$\bx,\bx'\in\RR^n$,
$\blmbd,\blmbd'\in\RR^m$,
$\bz\in\RR^n$, and
$\bmu\in\RR^m$, it holds that 
\begin{align}
    &\bigl\|
    \nabla_{\bx}\mathcal{P}_s(\bx,\blmbd,\bz,\bmu)
    -
    \nabla_{\bx}\mathcal{P}_s(\bx',\blmbd',\bz,\bmu)
    \bigr\|
    \le
    L_x\|\bx-\bx'\|
    +
    G_{\calA}\|\blmbd-\blmbd'\|,
    \label{eq:payoff-x-smoothness}
    \\
    &\bigl\|
    \nabla_{\blmbd}\mathcal{P}_s(\bx,\blmbd,\bz,\bmu)
    -
    \nabla_{\blmbd}\mathcal{P}_s(\bx',\blmbd',\bz,\bmu)
    \bigr\|
    \le
    G_{\calA}\|\bx-\bx'\|
    +
    L_\lambda(s)\|\blmbd-\blmbd'\|
    \label{eq:payoff-lambda-smoothness}.
\end{align}
In particular, $\mathcal P_s$ is $L_x$-smooth in $\bx$ and $L_\lambda(s)$-smooth in $\blmbd$.
\end{proposition}
 
\subsection{Value Functions and Lyapunov Function}
\label{subsec:value-functions}

Fix $s>0$ and $(\bz,\bmu)$. By Proposition~\ref{prop:curvature-smoothness},
$\mathcal P_s(\cdot,\cdot;\bz,\bmu)$ is strongly convex in $\bx$ and strongly
concave in $\blmbd$. Hence the following value functions are well defined,
and the minimizer and maximizer below are unique:
\begin{align}
    d_s(\blmbd;\bz,\bmu)
    &:= \min_{\bx\in\RR^n}\mathcal P_s(\bx,\blmbd;\bz,\bmu),
    &
    \bx_s(\blmbd;\bz,\bmu)
    &:= \argmin_{\bx\in\RR^n}\mathcal P_s(\bx,\blmbd;\bz,\bmu),
    \label{eq:dual-value}\\
    p_s(\bz,\bmu)
    &:= \max_{\blmbd\in\RR^m} d_s(\blmbd;\bz,\bmu),
    &
    \blmbd_s^\star(\bz,\bmu)
    &:= \argmax_{\blmbd\in\RR^m} d_s(\blmbd;\bz,\bmu).
    \label{eq:saddle-value}
\end{align}
Strong convexity--strong concavity also yields strong duality,
\begin{equation}
    p_s(\bz,\bmu)
    = \min_{\bx\in\RR^n}\max_{\blmbd\in\RR^m}
      \mathcal P_s(\bx,\blmbd;\bz,\bmu),
    \label{eq:minimax-equality}
\end{equation}
and the unique saddle point is
$(\bx_s^\star(\bz,\bmu),\blmbd_s^\star(\bz,\bmu))$, where
\[
    \bx_s^\star(\bz,\bmu)
    :=\bx_s\bigl(\blmbd_s^\star(\bz,\bmu);\bz,\bmu\bigr).
\]

\blue{Following the three-value Lyapunov construction for smoothed nonconvex--concave minimax methods~\cite{zhang2020smoothed}, and its perturbed-smoothed extension~\cite{li2026smoothing}, we adapt the construction to the reduced augmented-Lagrangian setting and define}
\begin{equation}
\begin{aligned}
    \Psi_s(\bx,\blmbd;\bz,\bmu)
    &:={}
    \mathcal P_s(\bx,\blmbd;\bz,\bmu)
    -2d_s(\blmbd;\bz,\bmu)
    +2p_s(\bz,\bmu) \\
    &= \underbrace{\mathcal P_s(\bx,\blmbd;\bz,\bmu)-d_s(\blmbd;\bz,\bmu)}_{\text{primal gap at fixed }\blmbd}
     + \underbrace{p_s(\bz,\bmu)-d_s(\blmbd;\bz,\bmu)}_{\text{dual gap}}
     + p_s(\bz,\bmu).
\end{aligned}
\label{eq:lyapunov-decomposition}
\end{equation}
Both gaps are nonnegative, so $\Psi_s\ge p_s$.

\paragraph{Iteration-$k$ shorthand.}
At iteration $k$ we evaluate the above objects at $(s,\bmu)=(s_k,\bmu^k)$,
replace the subscript $s$ by $k$, and suppress $\bmu^k$. For example,
$\mathcal P_k(\bx,\blmbd;\bz):=\mathcal P_{s_k}(\bx,\blmbd;\bz,\bmu^k)$ and
$d_k(\blmbd;\bz):=d_{s_k}(\blmbd;\bz,\bmu^k)$; the quantities
$p_k$, $\bx_k$, $\blmbd_k^\star$, $\bx_k^\star$, and $\Psi_k$ are defined
analogously. We also write
$\Delta_{\bx}^{k+1}:=\bx^{k+1}-\bx^k$, and define
$\Delta_{\blmbd}^{k+1}$, $\Delta_{\bz}^{k+1}$, and $\Delta_{\bmu}^{k+1}$
in the same way.

\subsection{Basic One-Step Estimate}
\label{subsec:basic-one-step}

Throughout this subsection, we assume that 
\begin{equation}
    s_{k+1}\ge s_k>0,\qquad
    0 < \tau_k\le s_k,\qquad
    0<\beta_k\le1,\qquad
    0\le\alpha_k\le2.
    \label{eq:basic-parameter-condition}
\end{equation} 
The following lemma gives a basic one-step descent estimate of the potential function $\mathcal P$ from iteration $k$ to $k+1$.

\begin{lemma}
\label{lem:basic-one-step}
Suppose Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold.
Let the iterates be generated by Algorithm~\ref{alg:ddlalm} with the
parameters satisfying \eqref{eq:basic-parameter-condition}.  Then, for
every $k\ge0$, the following holds:
\begin{equation}
\label{eq:basic-one-step}
\begin{aligned}
    &\mathcal P_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
    -\mathcal P_k(\bx^k,\blmbd^k;\bz^k) \\
    \le\;&
    -\left(\frac{1}{\eta_k}-\frac{L_x}{2}\right)
      \|\Delta_{\bx}^{k+1}\|^2
    +\left(\frac{1}{\tau_k}-\frac{1}{2s_k}\right)
      \|\Delta_{\blmbd}^{k+1}\|^2 \\
    &-\frac{\alpha_k(2-\alpha_k)}{2\gamma}
      \|\bx^{k+1}-\bz^k\|^2
    +\frac{\beta_k(2-\beta_k)}{2s_k}
      \|\blmbd^{k+1}-\bmu^k\|^2 +\frac{1}{2}
      \left(\frac{1}{s_k}-\frac{1}{s_{k+1}}\right)
      \|\blmbd^{k+1}-\bmu^{k+1}\|^2 .
\end{aligned}
\end{equation}
\end{lemma}


As shown in Lemma~\ref{lem:basic-one-step}, the positive ascent term
generated by the $\blmbd$-update prevents the potential
$\mathcal P_k$ from yielding a direct descent estimate. We therefore relate
this term to the one-step increase of the dual value function, as developed
in Lemma~\ref{lem:dual-value-gain}.  This relation allows us to construct a Lyapunov function with a descent property. Before proceeding, we establish several regularity properties of the value functions \(d_k\) and \(p_k\), with their gradient formulas following from Danskin's theorem.
\begin{lemma}
\label{lem:v2-value-regularity}
Under the conditions of Proposition~\ref{prop:curvature-smoothness}, for all $k\ge0$, the following properties hold:
\begin{enumerate}[label=(\roman*)]
\item
For every \(\bz \in \RR^n \), the minimizer
\(\bx_k(\blmbd;\bz)\) is unique and
\(\frac{G_{\cal A}}{\mu_\gamma}\)-Lipschitz continuous in
\(\blmbd\).

\item for every \(\bz\in\RR^n\),
\(d_k(\cdot;\bz)\) is \(1/s_k\)-strongly concave and
\(L_{d,k}\)-smooth, with gradient
\[
    \nabla_{\blmbd}d_k(\blmbd;\bz)
    =
    \nabla_{\blmbd}\mathcal P_k
    \bigl(\bx_k(\blmbd;\bz),\blmbd;\bz\bigr),
\]
where \(L_{d,k}:=\frac1\rho+\frac1{s_k}+\frac{G_{\cal A}^2}{\mu_\gamma}\).

\item
The function \(p_k(z)\) is continuously differentiable, with gradient
\[
    \nabla p_k(\bz)
    =
    \frac1\gamma
    \bigl(\bz-\bx_k^\star(\bz)\bigr),
\]
and has a \(1/\gamma\)-Lipschitz continuous gradient.
\end{enumerate}
\end{lemma}

The following lemma controls the dual ascent term.
\begin{lemma}
\label{lem:dual-value-gain}
Suppose that the conditions of
Lemma~\ref{lem:v2-value-regularity} hold, and let the iterates
be generated by Algorithm~\ref{alg:ddlalm} with parameters satisfying
\eqref{eq:basic-parameter-condition}.
For every $k\ge 0$, assume that
\begin{equation}
\label{eq:dual-gain-condition}
    \frac{1}{\tau_k}+\frac{1}{2s_k} \ge 2L_{d,k}.
\end{equation}
Then the following holds
\begin{equation*}
\begin{aligned}
& \left(\frac{1}{\tau_k}-\frac{1}{2s_k}\right)
  \|\Delta_{\boldsymbol{\lambda}}^{k+1}\|^2 \\
& \le 2\!\bigl[
    d_k(\boldsymbol{\lambda}^{k+1};\mathbf{z}^k)
    -d_k(\boldsymbol{\lambda}^k;\mathbf{z}^k)
  \bigr]
  -\frac{L_{d,k}}{2}
    \|\Delta_{\boldsymbol{\lambda}}^{k+1}\|^2
  +\frac{4G_{\mathcal A}^2}{L_{d,k}\mu_\gamma}
  \left[
    \mathcal P_k(
      \mathbf{x}^{k+1},\boldsymbol{\lambda}^k;\mathbf{z}^k)
    -d_k(\boldsymbol{\lambda}^k;\mathbf{z}^k)
  \right].
\end{aligned}
\end{equation*}
Moreover, it holds that
\begin{equation}
\begin{aligned}
&\mathcal P_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
-\mathcal P_k(\bx^k,\blmbd^k;\bz^k)
\\
&\le
-\left[
\frac1{\eta_k}-\frac{L_x}{2}
-
\frac{2G_{\mathcal A}^2(1/\eta_k+L_x)^2}
{L_{d,k}\mu_\gamma^2}
\right]
\|\Delta_{\bx}^{k+1}\|^2
-\frac{L_{d,k}}2\|\Delta_{\blmbd}^{k+1}\|^2 \\
&\qquad
+2\bigl[d_k(\blmbd^{k+1};\bz^k)-d_k(\blmbd^k;\bz^k)\bigr]
-\frac{\alpha_k(2-\alpha_k)}{2\gamma}
\|\bx^{k+1}-\bz^k\|^2 \\
&\qquad
+\frac{\beta_k(2-\beta_k)}{2s_k}
\|\blmbd^{k+1}-\bmu^k\|^2
+\frac12\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
\|\blmbd^{k+1}-\bmu^{k+1}\|^2.
\end{aligned}
\label{eq:dual-value-full-step}
\end{equation}
\end{lemma}

The potential $\mathcal P_k$ increases along the dual update, so its variation alone does not capture progress in
both variables. We therefore seek a merit function that accounts for both
primal suboptimality and progress in dual maximization. To account for
these two effects, consider the partial potential
\begin{equation}
\label{eq:partial-potential}
    \mathcal P_k-cd_k
    =(\mathcal P_k-d_k)-(c-1)d_k,
    \qquad c>1.
\end{equation}
The first term measures the primal optimality gap, while the second term
decreases as the dual value increases. In the following analysis, we set
$c=2$, consistent with the coefficient of the dual-value increment in
\eqref{eq:dual-value-full-step}.

\begin{lemma}
\label{lem:partial-lyapunov-z-step}
Under the conditions of Lemma~\ref{lem:dual-value-gain}, for every $k\ge0$, the following holds
\begin{equation}
\begin{aligned}
&\Big[
\mathcal P_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
-2d_k(\blmbd^{k+1};\bz^{k+1})
\Big] -
\Big[
\mathcal P_k(\bx^k,\blmbd^k;\bz^k)
-2d_k(\blmbd^k;\bz^k)
\Big]\\
\le\,&
-\left[
\frac1{\eta_k}-\frac{L_x}{2}
-\frac{2G_{\mathcal A}^2(1/\eta_k+L_x)^2}
{L_{d,k}\mu_\gamma^2}
\right]
\|\Delta_{\bx}^{k+1}\|^2 -\frac{L_{d,k}}{2}\|\Delta_{\blmbd}^{k+1}\|^2 \\
&\qquad
+\frac{\alpha_k(2-\alpha_k)}{2\gamma}
\|\bx^{k+1}-\bz^k\|^2
-\frac{2\alpha_k}{\gamma}
\left\langle
\bx^{k+1}
-\bx_k(\blmbd^{k+1};\bz^{k+1}),
\bx^{k+1}-\bz^k
\right\rangle  \\
&\qquad
+\frac{\beta_k(2-\beta_k)}{2s_k}
\|\blmbd^{k+1}-\bmu^k\|^2
+\frac12\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
\|\blmbd^{k+1}-\bmu^{k+1}\|^2.
\end{aligned}
\label{eq:partial-lyapunov-z-step}
\end{equation}
\end{lemma}
Equation~\eqref{eq:partial-lyapunov-z-step} gives the one-step estimate
for the partial potential in \eqref{eq:partial-potential} with $c=2$.
The $\bx$- and
$\blmbd$-updates contribute the first two terms on the right-hand side.
The $\bz$-update, however, produces a positive quadratic term and an
indefinite inner-product term. Hence, the partial potential does not yet
provide a sufficient descent estimate for the full iteration.

At an exact saddle point of
\(\min_{\bx}\max_{\blmbd}\mathcal P_k(\bx,\blmbd;\bz)\),
strong duality gives
\[
    \mathcal P_k\bigl(\bx_k^\star(\bz),
    \blmbd_k^\star(\bz);\bz\bigr)
    =d_k\bigl(\blmbd_k^\star(\bz);\bz\bigr)
    =p_k(\bz),
\]
and therefore
\[
    \mathcal P_k\bigl(\bx_k^\star(\bz),
    \blmbd_k^\star(\bz);\bz\bigr)
    -2d_k\bigl(\blmbd_k^\star(\bz);\bz\bigr)
    =-p_k(\bz).
\]
Consequently, any $\bz$-update that decreases the saddle value $p_k$
increases the partial potential $\mathcal P_k-2d_k$. This motivates us
to examine whether an additional value-function term can restore a descent
estimate.

We therefore add the value-function increment to the partial potential in
\eqref{eq:partial-potential}:
$2[p_k(\bz^{k+1})-p_k(\bz^k)]$ and combine the two estimates directly.
The resulting one-step bound is stated next.

\begin{lemma}
\label{lem:z-step-p}
Under the conditions of Lemma~\ref{lem:v2-value-regularity} and
$0\le\alpha_k\le2$, the $\bz$-update satisfies
\begin{equation}
\begin{aligned}
p_k(\bz^{k+1})-p_k(\bz^k)
\le\,
-\frac{\alpha_k(2-\alpha_k)}{2\gamma}
\|\bx^{k+1}-\bz^k\|^2 +
\frac{\alpha_k}{\gamma}
\left\langle
\bx^{k+1}-\bx_k^\star(\bz^k),
\bx^{k+1}-\bz^k
\right\rangle.
\end{aligned}
\label{eq:z-step-p}
% \notag
\end{equation}
\end{lemma}

\begin{lemma}
\label{lem:augmented-one-step}
Under the conditions of Lemma~\ref{lem:partial-lyapunov-z-step}, for every
$k\ge0$,
\begin{equation}
\begin{aligned}
&\Big[
\mathcal P_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
-2d_k(\blmbd^{k+1};\bz^{k+1})
+2p_k(\bz^{k+1})
\Big] \\
&\quad-
\Big[
\mathcal P_k(\bx^k,\blmbd^k;\bz^k)
-2d_k(\blmbd^k;\bz^k)
+2p_k(\bz^k)
\Big] \\
&\le
-\left[
\frac1{\eta_k}-\frac{L_x}{2}
-\frac{2G_{\mathcal A}^2(1/\eta_k+L_x)^2}
{L_{d,k}\mu_\gamma^2}
\right]
\|\Delta_{\bx}^{k+1}\|^2
-\frac{L_{d,k}}{2}\|\Delta_{\blmbd}^{k+1}\|^2 \\
&\qquad
-\frac{\alpha_k(2-\alpha_k)}{2\gamma}
\|\bx^{k+1}-\bz^k\|^2
+\frac{2\alpha_k}{\gamma}
\left\langle
\bx_k(\blmbd^{k+1};\bz^{k+1})
-\bx_k^\star(\bz^k),
\bx^{k+1}-\bz^k
\right\rangle \\
&\qquad
+\frac{\beta_k(2-\beta_k)}{2s_k}
\|\blmbd^{k+1}-\bmu^k\|^2
+\frac12\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
\|\blmbd^{k+1}-\bmu^{k+1}\|^2.
\end{aligned}
\label{eq:augmented-one-step}
\end{equation}
\end{lemma}

Two gaps remain before $\Phi_k$ admits a genuine descent inequality: the augmented estimate still carries an indefinite cross term and positive multiplier/center-increment terms, and its left-hand side compares $d_k, p_k$ at both endpoints rather than the true $k \to k+1$ potential difference, which requires $d_{k+1}, p_{k+1}$. 

For that, we establish boundedness of the multipliers directly, rather than assuming compactness a priori: both multipliers are controlled by the AL structure and the inner residual, not by projection onto a compact set.
\begin{lemma}
\label{lem:current-multiplier-boundedness}
Suppose Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold.
Let the iterates be generated by Algorithm~\ref{alg:ddlalm} with
\eqref{eq:basic-parameter-condition} and
\begin{equation}
\label{eq:multiplier-boundedness-condition}
    \tau_k\left(\frac1\rho+\frac1{s_k}\right)\le1.
\end{equation}
Define the constant independent of $k$
\[
    M:=\max\{\|\blmbd^0\|,\|\bmu^0\|,L_h\}.
\]
Then, for every $k\ge0$, the following holds
\[
    \|\blmbd^k\|\le M,
    \qquad
    \|\bmu^k\|\le M.
\]
Moreover, for every $k\ge0$ and every $\bz\in\RR^n$, it holds that
\[
    \|\blmbd_k^\star(\bz)\|\le M.
\]
\end{lemma}

\begin{lemma}
\label{lem:saddle-center-stability}
Under the conditions of Proposition~\ref{prop:curvature-smoothness}, for every $k\ge0$, $\blmbd\in\RR^m$, and $\bz\in\RR^n$, it holds that
\begin{equation}
\label{eq:current-saddle-stability}

\frac{2}{\mu_\gamma}
\bigl[p_k(\bz)-d_k(\blmbd;\bz)\bigr].
\end{equation}

\begin{equation}
\label{eq:current-center-stability}

\|\bx_k(\blmbd;\bz')-\bx_k(\blmbd;\bz)\|,

\bigr\}
\le\frac{1}{\gamma\mu_\gamma}\|\bz'-\bz\|.
\end{equation}
\end{lemma}

We now establish an approximate descent inequality for
\Cref{alg:ddlalm} with respect to the Lyapunov function defined in \eqref{eq:lyapunov-decomposition}. We assume 
the parameters are chosen as

\label{eq:parameter-choice}
\begin{gathered}
    0<\gamma<\frac{1}{2L_{\calD,\bx}},\qquad
    0<s_k\le s_{k+1},\qquad
    0<\beta_k\le1,\qquad
    0<\eta_k\le\frac{1}{2L_x},\\

\end{gathered}
\end{equation}

Since $L_{d,k}\ge\rho^{-1}+s_k^{-1}$, \eqref{eq:parameter-choice} implies the conditions of Proposition~\ref{prop:curvature-smoothness},
Lemma~\ref{lem:current-multiplier-boundedness},
\eqref{eq:basic-parameter-condition},
\eqref{eq:dual-gain-condition}, and
\eqref{eq:multiplier-boundedness-condition}.

\begin{lemma}
\label{lem:moving-three-value-descent}
\blue{Suppose that Assumptions~\ref{ass:level-boundedness},
\ref{ass:smoothness}, and~\ref{ass:h} hold, and let}
$\{(\bx^k,\blmbd^k,\bz^k,\bmu^k)\}$ be generated by \Cref{alg:ddlalm}
with parameters satisfying~\eqref{eq:parameter-choice}. Then for every
$k\ge0$:
\begin{enumerate}[label=(\alph*)]
    \item for all $\bx\in\RR^n$, $\blmbd\in\RR^m$, and $\bz\in\RR^n$,
    \begin{equation}
    \label{eq:current-lyapunov-lower-bound}
        \Psi_k(\bx,\blmbd;\bz)\ge p_k(\bz)
        \ge\inf_{\bx\in\RR^n}F(\bx)-\frac{(L_h+M)^2}{2s_0};
    \end{equation}
    \item the iterates satisfy
    \begin{equation}
    \label{eq:full-lyapunov-descent}
    \begin{aligned}
        &\Psi_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
         -\Psi_k(\bx^k,\blmbd^k;\bz^k)\\
        &\qquad\le
         -\frac1{4\eta_k}\|\Delta_{\bx}^{k+1}\|^2
         -\frac1{8\tau_k}\|\Delta_{\blmbd}^{k+1}\|^2
         -\frac{\alpha_k}{2\gamma}\|\bx^{k+1}-\bz^k\|^2
         +\delta_k,
    \end{aligned}
    \end{equation}
    where
    $\delta_k:=10M^2\bigl(s_k^{-1}-s_{k+1}^{-1}\bigr)+20M^2\beta_k/s_k$.
\end{enumerate}
\end{lemma}

\begin{proof}[Proof of the curvature part of Proposition~\ref{prop:curvature-smoothness}]
For fixed $\blmbd$, the reduced payoff satisfies
\[
    \nabla_{\bx}\mathcal D_\rho(\bx,\blmbd)
    =
    \nabla f(\bx)+J_{\calA}(\bx)^\top\bq(\bx,\blmbd),
\]
where
\[
    \bq(\bx,\blmbd)
    =
    \operatorname{prox}_{\rho h^*}
    \bigl(\blmbd+\rho\calA(\bx)\bigr).
\]

The nonexpansiveness of the proximal mapping, the bounded Jacobian, and
the Lipschitz Jacobian imply
\[
    \|\bq(\bx,\blmbd)-\bq(\bx',\blmbd)\|
    \le \rho G_{\calA}\|\bx-\bx'\|.
\]
Assumption~\ref{ass:h} gives
\(\|\bq(\bx,\blmbd)\|\le L_h\).  Therefore
\[
    \|\nabla_{\bx}\mathcal D_\rho(\bx,\blmbd)
    -\nabla_{\bx}\mathcal D_\rho(\bx',\blmbd)\|
    \le L_{\calD,\bx}\|\bx-\bx'\|.
\]

Thus $\mathcal D_\rho(\cdot,\blmbd)$ is
$L_{\calD,\bx}$-weakly convex.  Adding
$\frac{1}{2\gamma}\|\bx-\bz\|^2$ gives
$\mu_\gamma$-strong convexity in $\bx$.

For fixed $(\bx,\bz,\bmu)$, $\mathcal D_\rho(\bx,\cdot)$ is concave in
$\blmbd$ because it is the pointwise infimum of affine functions of
$\blmbd$.  The term
$-\frac{1}{2s}\|\blmbd-\bmu\|^2$ is $1/s$-strongly concave.  Hence
$\mathcal P_s$ is $1/s$-strongly concave in $\blmbd$.
\end{proof}

\begin{proof}[Proof of the smoothness part of Proposition~\ref{prop:curvature-smoothness}]
The $\bx$-block bound follows from the Lipschitz estimate for
$\nabla_{\bx}\mathcal D_\rho$ in the proof of
Proposition~\ref{prop:curvature-smoothness} and the quadratic term
$\frac{1}{2\gamma}\|\bx-\bz\|^2$.

For the $\blmbd$-block, the identity
\[
    \nabla_{\blmbd}\mathcal P_s(\bx,\blmbd;\bz,\bmu)
    =
    \frac{1}{\rho}
    \bigl(\bq(\bx,\blmbd)-\blmbd\bigr)
    -\frac{1}{s}(\blmbd-\bmu)
\]
and the nonexpansiveness of
$\operatorname{prox}_{\rho h^*}$ give
$L_\lambda(s)=1/\rho+1/s$.
The same estimate, together with
\[
    \|\calA(\bx)-\calA(\bx')\|
    \le G_{\calA}\|\bx-\bx'\|,
\]
gives both cross Lipschitz bounds with constant $G_{\calA}$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:dual-value-gain}]
Set
\[
    g^k
    :=
    \nabla_{\blmbd}\mathcal P_k
    (\bx^{k+1},\blmbd^k,\bz^k),
    \qquad
    \Delta_{\blmbd}^{k+1}=\tau_k g^k,
\]
where the second identity follows from the dual update.
By the $L_{d,k}$-smoothness of $d_k(\cdot;\bz^k)$,
\[
\begin{aligned}
&d_k(\blmbd^{k+1};\bz^k)-d_k(\blmbd^k;\bz^k) \\
&\quad\ge
\left\langle
\nabla_{\blmbd}d_k(\blmbd^k;\bz^k),
\Delta_{\blmbd}^{k+1}
\right\rangle
-\frac{L_{d,k}}{2}
\|\Delta_{\blmbd}^{k+1}\|^2.
\end{aligned}
\]
Danskin's theorem and Lemma~\ref{lem:v2-value-regularity} give
\[
\nabla_{\blmbd}d_k(\blmbd^k;\bz^k)
=
\nabla_{\blmbd}\mathcal P_k
\bigl(\bx_k(\blmbd^k;\bz^k),\blmbd^k,\bz^k\bigr).
\]
The cross-Lipschitz estimate in
Proposition~\ref{prop:curvature-smoothness} therefore yields
\[
\begin{aligned}
&\left\langle
\nabla_{\blmbd}d_k(\blmbd^k;\bz^k),
\Delta_{\blmbd}^{k+1}
\right\rangle \\
&\quad=
\tau_k\left\langle
\nabla_{\blmbd}\mathcal P_k
\bigl(\bx_k(\blmbd^k;\bz^k),\blmbd^k,\bz^k\bigr),g^k
\right\rangle \\
&\quad\ge
\tau_k\|g^k\|^2
-\tau_kG_{\cal A}
\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|\|g^k\|.
\end{aligned}
\]
Using
\[
    \tau_k\|g^k\|^2
    =\frac{1}{\tau_k}\|\Delta_{\blmbd}^{k+1}\|^2,
    \qquad
    \tau_k\|g^k\|
    =\|\Delta_{\blmbd}^{k+1}\|,
\]
we obtain
\begin{equation}
\begin{aligned}
&d_k(\blmbd^{k+1};\bz^k)-d_k(\blmbd^k;\bz^k) \\
&\quad\ge
\left(\frac{1}{\tau_k}-\frac{L_{d,k}}{2}\right)
\|\Delta_{\blmbd}^{k+1}\|^2
-G_{\cal A}\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|
\|\Delta_{\blmbd}^{k+1}\|.
\end{aligned}
\label{eq:dual-value-gain-appendix}
\end{equation}

Let
\[
    b_k:=
    \frac{1}{\tau_k}+\frac{1}{2s_k}-L_{d,k}.
\]
Multiplying \eqref{eq:dual-value-gain-appendix} by $2$ and rearranging gives
\[
\begin{aligned}
&\left(\frac{1}{\tau_k}-\frac{1}{2s_k}\right)
\|\Delta_{\blmbd}^{k+1}\|^2 \\
&\quad\le
2\bigl[d_k(\blmbd^{k+1};\bz^k)-d_k(\blmbd^k;\bz^k)\bigr]
-b_k\|\Delta_{\blmbd}^{k+1}\|^2 \\
&\qquad+
2G_{\cal A}\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|
\|\Delta_{\blmbd}^{k+1}\|.
\end{aligned}
\]
The assumed parameter condition implies $b_k\ge L_{d,k}>0$. Young's inequality
with parameter $L_{d,k}$ gives
\[
2G_{\cal A}\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|
\|\Delta_{\blmbd}^{k+1}\|
\le
\frac{L_{d,k}}{2}\|\Delta_{\blmbd}^{k+1}\|^2
+\frac{2G_{\cal A}^2}{L_{d,k}}
\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|^2.
\]
Since $b_k\ge L_{d,k}$, we have
\[
    -b_k\|\Delta_{\blmbd}^{k+1}\|^2
    +\frac{L_{d,k}}{2}\|\Delta_{\blmbd}^{k+1}\|^2
    \le
    -\frac{L_{d,k}}{2}\|\Delta_{\blmbd}^{k+1}\|^2.
\]
Finally, the $\mu_\gamma$-strong convexity of
$\mathcal P_k(\cdot,\blmbd^k;\bz^k)$ gives
\[
\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|^2
\le
\frac{2}{\mu_\gamma}
\left[
\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k)
-d_k(\blmbd^k;\bz^k)
\right].
\]
Substituting this estimate proves the first inequality.
\end{proof}

\begin{proof}[Proof of the second estimate in Lemma~\ref{lem:dual-value-gain}]
Fix $k\ge0$. By Proposition~\ref{prop:curvature-smoothness},
$\mathcal P_k(\cdot,\blmbd^k;\bz^k)$ is
$\mu_\gamma$-strongly convex. Applying the strong-convexity inequality
at $\bx^{k+1}$ to the minimizer $\bx_k(\blmbd^k;\bz^k)$ gives
\[
\begin{aligned}
&\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k)-d_k(\blmbd^k;\bz^k) \\
&\quad\le
\left\langle
\nabla_{\bx}\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k),
\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)
\right\rangle \\
&\qquad
-\frac{\mu_\gamma}{2}
\|\bx^{k+1}-\bx_k(\blmbd^k;\bz^k)\|^2 \\
&\quad\le
\frac{1}{2\mu_\gamma}
\|\nabla_{\bx}\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k)\|^2,
\end{aligned}
\]
where the last inequality follows from Young's inequality.
The $\bx$-update and the $L_x$-Lipschitz continuity of
$\nabla_{\bx}\mathcal P_k(\cdot,\blmbd^k;\bz^k)$ imply
\[
\begin{aligned}
&\|\nabla_{\bx}\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k)\| \\
&\quad\le
\|\nabla_{\bx}\mathcal P_k(\bx^k,\blmbd^k;\bz^k)\|
+L_x\|\Delta_{\bx}^{k+1}\| \\
&\quad=
\left(\frac{1}{\eta_k}+L_x\right)
\|\Delta_{\bx}^{k+1}\|.
\end{aligned}
\]
Combining these estimates yields
\begin{equation}
\label{eq:primal-gap-control}
\mathcal P_k(\bx^{k+1},\blmbd^k;\bz^k)
-d_k(\blmbd^k;\bz^k)
\le
\frac{(1/\eta_k+L_x)^2}{2\mu_\gamma}
\|\Delta_{\bx}^{k+1}\|^2.
\end{equation}
Substituting \eqref{eq:primal-gap-control} into the first inequality of
Lemma~\ref{lem:dual-value-gain} gives a bound on the dual ascent term,
where the coefficient of
$\|\Delta_{\bx}^{k+1}\|^2$ becomes
\[
\frac{4G_{\mathcal A}^2}{L_{d,k}\mu_\gamma}
\frac{(1/\eta_k+L_x)^2}{2\mu_\gamma}
=
\frac{2G_{\mathcal A}^2(1/\eta_k+L_x)^2}
{L_{d,k}\mu_\gamma^2}.
\]
Substitute the resulting bound on
$(1/\tau_k-1/(2s_k))\|\Delta_{\blmbd}^{k+1}\|^2$
into \eqref{eq:basic-one-step} and combine the coefficients of
$\|\Delta_{\bx}^{k+1}\|^2$. The $\bz$-step term and both parameter-drift
terms remain unchanged. This proves \eqref{eq:dual-value-full-step}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:partial-lyapunov-z-step}]
Subtract
$2[d_k(\blmbd^{k+1};\bz^k)-d_k(\blmbd^k;\bz^k)]$
from both sides of \eqref{eq:dual-value-full-step}. To replace
$d_k(\blmbd^{k+1};\bz^k)$ on the left by
$d_k(\blmbd^{k+1};\bz^{k+1})$, add
$-2[d_k(\blmbd^{k+1};\bz^{k+1})-d_k(\blmbd^{k+1};\bz^k)]$
to both sides.
By the minimum-value definition,
\[
d_k(\blmbd^{k+1};\bz^k)
\le
\mathcal P_k\bigl(\bx_k(\blmbd^{k+1};\bz^{k+1}),
\blmbd^{k+1};\bz^k\bigr).
\]
Subtracting this inequality from the corresponding equality at
$\bz^{k+1}$ and expanding the quadratic terms yields
\[
\begin{aligned}
&d_k(\blmbd^{k+1};\bz^{k+1})-d_k(\blmbd^{k+1};\bz^k) \\
&\quad\ge
\frac1\gamma\left\langle
\bz^k-\bx_k(\blmbd^{k+1};\bz^{k+1}),
\Delta_{\bz}^{k+1}\right\rangle
+\frac1{2\gamma}\|\Delta_{\bz}^{k+1}\|^2 \\
&\quad=
-\frac{\alpha_k(2-\alpha_k)}{2\gamma}
\|\bx^{k+1}-\bz^k\|^2 \\
&\qquad+
\frac{\alpha_k}{\gamma}\left\langle
\bx^{k+1}-\bx_k(\blmbd^{k+1};\bz^{k+1}),
\bx^{k+1}-\bz^k\right\rangle.
\end{aligned}
\]
Multiply this inequality by $-2$ and substitute it into the preceding
estimate. Its squared-step term combines with the $\bz$-step term in
\eqref{eq:dual-value-full-step} to give
$\alpha_k(2-\alpha_k)\|\bx^{k+1}-\bz^k\|^2/(2\gamma)$.
Both parameter-drift terms remain unchanged, proving
\eqref{eq:partial-lyapunov-z-step}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:z-step-p}]
By Lemma~\ref{lem:v2-value-regularity}, $p_k$ has a
$1/\gamma$-Lipschitz continuous gradient and
\[
    \nabla p_k(\bz^k)
    =
    \frac1\gamma\bigl(\bz^k-\bx_k^\star(\bz^k)\bigr).
\]
The $\bz$-update gives
\[
    \bz^{k+1}-\bz^k
    =\alpha_k(\bx^{k+1}-\bz^k).
\]
Applying the smoothness inequality for $p_k$ and expanding the inner
product yields \eqref{eq:z-step-p}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:augmented-one-step}]
By Lemma~\ref{lem:z-step-p},
\begin{equation}
\begin{aligned}
2\bigl[p_k(\bz^{k+1})-p_k(\bz^k)\bigr]
\le{}&
-\frac{\alpha_k(2-\alpha_k)}{\gamma}
\|\bx^{k+1}-\bz^k\|^2 \\
&+
\frac{2\alpha_k}{\gamma}
\left\langle
\bx^{k+1}-\bx_k^\star(\bz^k),
\bx^{k+1}-\bz^k
\right\rangle.
\end{aligned}
\end{equation}
Adding this inequality to
\eqref{eq:partial-lyapunov-z-step}, the quadratic terms combine as
\[
\frac{\alpha_k(2-\alpha_k)}{2\gamma}
-\frac{\alpha_k(2-\alpha_k)}{\gamma}
=
-\frac{\alpha_k(2-\alpha_k)}{2\gamma}.
\]
The two inner products combine into
\[
\frac{2\alpha_k}{\gamma}
\left\langle
\bx_k(\blmbd^{k+1};\bz^{k+1})
-\bx_k^\star(\bz^k),
\bx^{k+1}-\bz^k
\right\rangle.
\]
Substitution into \eqref{eq:partial-lyapunov-z-step} proves
\eqref{eq:augmented-one-step}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:current-multiplier-boundedness}]
The proximal optimality condition gives
$\bq(\bx,\blmbd)\in\partial h(\by(\bx,\blmbd))$, and hence
$\|\bq(\bx,\blmbd)\|\le L_h$. Substituting
$\calA(\bx)-\by(\bx,\blmbd)
=(\bq(\bx,\blmbd)-\blmbd)/\rho$ into the dual update yields
\[
\blmbd^{k+1}
=\left[1-\tau_k\left(\frac1\rho+\frac1{s_k}\right)\right]\blmbd^k
+\frac{\tau_k}{\rho}\bq(\bx^{k+1},\blmbd^k)
+\frac{\tau_k}{s_k}\bmu^k.
\]
The coefficients are nonnegative and sum to one. Together with
$\bmu^{k+1}=(1-\beta_k)\bmu^k+\beta_k\blmbd^{k+1}$, this proves
$\|\blmbd^k\|,\|\bmu^k\|\le M$ by induction from the definition of $M$.
The center update then gives
$\|\bmu^{k+1}-\bmu^k\|\le2M\beta_k$.

At the saddle point, the dual first-order condition gives
\[
\blmbd_k^\star(\bz)
=\frac{s_k}{\rho+s_k}
  \bq\bigl(\bx_k^\star(\bz),\blmbd_k^\star(\bz)\bigr)
+\frac{\rho}{\rho+s_k}\bmu^k.
\]
Since both vectors in this convex combination have norm at most $M$,
$\|\blmbd_k^\star(\bz)\|\le M$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:saddle-center-stability}]
Fix $k$ and $\bz$. Strong convexity gives the unique minimizer
$\bx_k(\blmbd;\bz)$ for each $\blmbd$. The finite, smooth,
$1/s_k$-strongly concave function $d_k(\cdot;\bz)$ tends to $-\infty$
as $\|\blmbd\|\to\infty$, so it has a unique maximizer.
The minimax equality and strong convexity in $\bx$ identify the unique
saddle pair and give
$\bx_k^\star(\bz)=\bx_k(\blmbd_k^\star(\bz);\bz)$.

The maximizing property at $\bx_k^\star(\bz)$ and strong convexity at
fixed $\blmbd$ imply
\[
\begin{aligned}
p_k(\bz)-d_k(\blmbd;\bz)
&\ge
\mathcal P_k(\bx_k^\star(\bz),\blmbd;\bz)
-\mathcal P_k(\bx_k(\blmbd;\bz),\blmbd;\bz) \\
&\ge\frac{\mu_\gamma}{2}
\|\bx_k(\blmbd;\bz)-\bx_k^\star(\bz)\|^2.
\end{aligned}
\]
This proves \eqref{eq:current-saddle-stability}.
For the center sensitivity of $\bx_k(\blmbd;\bz)$, add the
strong-convexity inequalities at the minimizers for $\bz$ and $\bz'$.
The only center-dependent term is $\|\bx-\bz\|^2/(2\gamma)$, so
\[
\mu_\gamma\|\bx_k(\blmbd;\bz')-\bx_k(\blmbd;\bz)\|^2
\le\frac1\gamma\left\langle
\bz'-\bz,\bx_k(\blmbd;\bz')-\bx_k(\blmbd;\bz)\right\rangle.
\]
For the saddle primal points, use the same argument for
$\max_{\blmbd}\mathcal P_k(\cdot,\blmbd;\bz)$, whose strong-convexity
modulus is also $\mu_\gamma$. Cauchy--Schwarz gives
\eqref{eq:current-center-stability} in both cases.
The latter argument also gives
\[
\left\langle\bz'-\bz,\bx_k^\star(\bz')-\bx_k^\star(\bz)\right\rangle
\ge\gamma\mu_\gamma
\|\bx_k^\star(\bz')-\bx_k^\star(\bz)\|^2.
\]
Together with the Danskin formula for $\nabla p_k$ and
$2\gamma\mu_\gamma>1$, this verifies the smoothness bound used above:
\[
\begin{aligned}
\gamma^2\|\nabla p_k(\bz')-\nabla p_k(\bz)\|^2
&\le\|\bz'-\bz\|^2
-(2\gamma\mu_\gamma-1)
\|\bx_k^\star(\bz')-\bx_k^\star(\bz)\|^2 \\
&\le\|\bz'-\bz\|^2.
\end{aligned}
\]
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:moving-three-value-descent}]
Let $u_{k+1}:=(\bx^{k+1},\blmbd^{k+1},\bz^{k+1})$ and define
\[
\widehat\Psi_{k+1|k}:=
\mathcal P_{k+1}(\bx^{k+1},\blmbd^{k+1};\bz^{k+1})
-2d_k(\blmbd^{k+1};\bz^{k+1})+2p_k(\bz^{k+1}).
\]
Then
\[
\Psi_{k+1}(u_{k+1})-\Psi_k(u_k)
=\widehat\Psi_{k+1|k}-\Psi_k(u_k)+R_k,
\]
where
\[
R_k:=2\bigl[d_k(\blmbd^{k+1};\bz^{k+1})
-d_{k+1}(\blmbd^{k+1};\bz^{k+1})\bigr]
+2\bigl[p_{k+1}(\bz^{k+1})-p_k(\bz^{k+1})\bigr].
\]

Apply Lemma~\ref{lem:augmented-one-step} to the first difference.
Decompose the indefinite term at $\bx_k^\star(\bz^{k+1})$ and apply
Lemma~\ref{lem:saddle-center-stability}, followed by Young's inequality.
The four bounds in \eqref{eq:parameter-choice} absorb the resulting
cross terms and leave
\[
\begin{aligned}
\widehat\Psi_{k+1|k}-\Psi_k(u_k)
\le{}&
-\frac1{4\eta_k}\|\Delta_{\bx}^{k+1}\|^2
-\frac1{8\tau_k}\|\Delta_{\blmbd}^{k+1}\|^2
-\frac{\alpha_k}{2\gamma}\|\bx^{k+1}-\bz^k\|^2 \\
&+\frac{\beta_k(2-\beta_k)}{2s_k}
\|\blmbd^{k+1}-\bmu^k\|^2
\\
&+\frac12\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
\|\blmbd^{k+1}-\bmu^{k+1}\|^2.
\end{aligned}
\]

It remains to bound $R_k$. Lemma~\ref{lem:current-multiplier-boundedness}
gives $\|\blmbd^j\|,\|\bmu^j\|,\|\blmbd_j^\star(\bz)\|\le M$ and
$\|\bmu^{k+1}-\bmu^k\|\le2M\beta_k$. Direct comparison of the two
quadratic damping terms gives, uniformly in $\bx$ and $\bz$,
\[
|d_{k+1}(\blmbd;\bz)-d_k(\blmbd;\bz)|
\le\Delta_k,
\qquad
|p_{k+1}(\bz)-p_k(\bz)|\le\Delta_k,
\]
where
\[
\Delta_k:=2M^2\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
+\frac{2M}{s_k}\|\bmu^{k+1}-\bmu^k\|.
\]
Consequently,
\[
|R_k|\le4\Delta_k
\le8M^2\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
+16M^2\frac{\beta_k}{s_k}.
\]
The two explicit positive terms above satisfy
\[
\frac{\beta_k(2-\beta_k)}{2s_k}
\|\blmbd^{k+1}-\bmu^k\|^2
\le4M^2\frac{\beta_k}{s_k},
\]
and
\[
\frac12\left(\frac1{s_k}-\frac1{s_{k+1}}\right)
\|\blmbd^{k+1}-\bmu^{k+1}\|^2
\le2M^2\left(\frac1{s_k}-\frac1{s_{k+1}}\right).
\]
Adding the estimates proves \eqref{eq:full-lyapunov-descent}.

For the stated schedule, the two error series satisfy
\[
\sum_{k=0}^\infty\left(\frac1{s_k}-\frac1{s_{k+1}}\right)=\frac1{c_s},
\qquad
\sum_{k=0}^\infty\frac{\beta_k}{s_k}
\le\frac{4\theta_\mu}{c_s},
\]
which makes the accumulated transport error finite. The lower bound
\eqref{eq:current-lyapunov-lower-bound} follows as follows. For any
$\blmbd\in\partial h(\calA(\bx))$, proximal optimality gives
$\by(\bx,\blmbd)=\calA(\bx)$ and hence
$\mathcal D_\rho(\bx,\blmbd)=F(\bx)$. Since
$\|\blmbd\|\le L_h$ and $\|\bmu^k\|\le M$,
\[
\max_{\blmbd}\mathcal P_k(\bx,\blmbd;\bz)
\ge F(\bx)+\frac1{2\gamma}\|\bx-\bz\|^2
-\frac{(L_h+M)^2}{2s_k}.
\]
Minimizing in $\bx$ and using $s_k\ge s_0$ proves
\eqref{eq:current-lyapunov-lower-bound}, because
$\Psi_k=(\mathcal P_k-d_k)+(p_k-d_k)+p_k\ge p_k$.
Summing \eqref{eq:full-lyapunov-descent} and using the error budget
then gives
\[
\begin{aligned}
&\sum_{k=0}^{N-1}\left[
\frac1{4\eta_k}\|\Delta_{\bx}^{k+1}\|^2
+\frac1{8\tau_k}\|\Delta_{\blmbd}^{k+1}\|^2
+\frac{\alpha_k}{2\gamma}\|\bx^{k+1}-\bz^k\|^2\right] \\
&\quad\le
\Psi_0(\bx^0,\blmbd^0;\bz^0)-\inf_{\bx\in\RR^n}F(\bx)
+\frac{(L_h+M)^2}{2s_0}
+\frac{(10+80\theta_\mu)M^2}{c_s},
\qquad N\ge1.
\end{aligned}
\]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
The schedule \eqref{eq:ddlalm-parameter-schedule} satisfies
\eqref{eq:parameter-choice}: $\eta_kL_x=1/2$,
$\tau_kL_{d,k}\le\tau_kL_{d,0}\le1/2$,
$\tau_kG_{\cal A}^2\le\mu_\gamma^2/(72L_x)
=\eta_k\mu_\gamma^2/36$, and the two bounds on $\alpha_k$ follow
from $s_k\ge s_0$ and its definition. Moreover,
$s_{k+1}\ge s_k>0$, $\eta_k,\tau_k>0$, and
$0<\beta_k\le1$. Since $L_{d,k}\ge1/\rho+1/s_k$,
the multiplier-boundedness condition also holds.
Level-boundedness and continuity of $F$ imply
$\inf_{\bx}F(\bx)>-\infty$. The master budget proved after
Lemma~\ref{lem:moving-three-value-descent} is therefore finite and
independent of $K$.

The tail has $\lfloor K/2\rfloor\ge K/3$ elements. Since the primal
and dual stepsizes are fixed and
$\alpha_k\ge\alpha_{K-1}=\Theta(K^{-1/3})$ on the tail, the master
budget and the definition of $\kappa$ give
\[
\|\Delta_{\bx}^{\kappa+1}\|^2=O(K^{-1}),\qquad
\|\Delta_{\blmbd}^{\kappa+1}\|^2=O(K^{-1}),\qquad
\|\bx^{\kappa+1}-\bz^\kappa\|^2=O(K^{-2/3}).
\]
The output certificate follows directly from
\eqref{eq:certificate-membership}. For stationarity, the $\bx$-update
and $L_{\calD,\bx}$-smoothness of $\mathcal D_\rho(\cdot,\blmbd^k)$ give
\[
\|\nabla f(\bx^{k+1})+J_{\cal A}(\bx^{k+1})^\top\bq^k\|
\le\left(\frac1{\eta_k}+L_x\right)\|\Delta_{\bx}^{k+1}\|
+\frac1\gamma\|\bx^{k+1}-\bz^k\|.
\]
For feasibility, the dual update gives
\[
{\cal A}(\bx^{k+1})-\by^{k+1}
=\frac1{\tau_k}\Delta_{\blmbd}^{k+1}
+\frac1{s_k}(\blmbd^k-\bmu^k),
\]
and hence
\[
\|{\cal A}(\bx^{k+1})-\by^{k+1}\|
\le\frac1{\tau_k}\|\Delta_{\blmbd}^{k+1}\|+\frac{2M}{s_k}.
\]
Square the last two bounds using $(a+b)^2\le2a^2+2b^2$ and use
$s_\kappa^{-2}=O(K^{-2/3})$. Both squared residuals are
$O(K^{-2/3})$, so their norms are $O(K^{-1/3})$. All constants depend
only on the problem data and fixed initialization and parameters, so
$K=O(\epsilon^{-3})$ suffices.
\end{proof}
