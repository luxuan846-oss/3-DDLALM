\section{Dual-Damping Linearized Augmented Lagrangian Method}
\label{sec:method}

This section develops a single-loop primal-dual algorithm for problem~\eqref{prob:main}.  


\subsection{Smoothed Perturbed augmented Lagrangian}
\label{subsec:dual-damping-reformulation}
We first transform Problem~\eqref{prob:main} into a constrained problem by introducing the auxiliary variable \(\by=\calA(\bx)\):
\begin{equation}\label{eq:split-problem}
    \min_{\bx\in\RR^n,\,\by\in\RR^m}
    \; f(\bx)+ h(\by)
    \quad\text{s.t.}\quad
    \calA(\bx)-\by=\zero,
\end{equation} whose augmented Lagrangian for~Problem \eqref{eq:split-problem} with penalty
\(\rho>0\) and multiplier \(\blmbd\in\RR^m\) is
\begin{equation}\label{eq:augmented-lagrangian}
    \calL_\rho(\bx,\by,\blmbd)
    :=
    f(\bx)+h(\by)
    +\langle \blmbd,\calA(\bx)-\by\rangle
    +\frac{\rho}{2}\|\calA(\bx)-\by\|^2.
\end{equation}

We first record the value obtained by eliminating the auxiliary variable
\(\by\) from the unregularized augmented Lagrangian.  Define the value function
\begin{equation}\label{eq:reduced-al-base}
    \begin{aligned}
        \mathcal D_\rho\bigl(\bx,\blmbd; \by(\bx,\blmbd)\bigr)
        &:=\mathcal L_\rho\bigl(\bx,\by(\bx,\blmbd),\blmbd\bigr) \\
        &=f(\bx)
        +M_{h/\rho}\!\left(\calA(\bx)+\frac{1}{\rho}\blmbd\right)
        -\frac{1}{2\rho}\|\blmbd\|^2,
    \end{aligned}
\end{equation}
where
\[
    M_{h/\rho}(v):=\min_{\by}\left\{
        h(\by)+\frac{\rho}{2}\|\by-v\|^2
    \right\}.
\]
The corresponding \(\by\) is given by
\begin{equation}\label{eq:rho-prox-map}
    \by (\bx,\blmbd)
    :=\operatorname{prox}_{h/\rho}
      \left(\calA(\bx)+\frac{1}{\rho}\blmbd\right).
\end{equation}
However, \(\mathcal D_\rho(\bx,\blmbd;\by(\bx,\blmbd))\) is generally nonconvex in \(\bx\)
and concave in \(\blmbd\). To regularize the two blocks, fix a primal
proximal parameter \(\gamma>0\) and a dual regularization parameter \(s>0\),
and define the smoothe, perturbed augmented Lagrangian as
\begin{equation}\label{eq:regularized-potential}
    \begin{aligned}
        \mathcal P_s(\bx,\blmbd;\bz,\bmu,\by(\bx,\blmbd) )
        :=\;&  \mathcal D_\rho\bigl(\bx,\blmbd;\by(\bx,\blmbd)\bigr)
        +\frac{1}{2\gamma}\|\bx-\bz\|^2
        -\frac{1}{2s}\|\blmbd-\bmu\|^2.
    \end{aligned}
\end{equation} 

\subsection{Algorithm Design}
\label{subsec:ddlalm-algorithm}
The method is formally presented in \Cref{alg:ddlalm}.

The gradients used in Algorithm~\ref{alg:ddlalm} are
\begin{equation}\label{eq:explicit-p-gradients}
    \begin{aligned}
        \nabla_{\bx}\mathcal P_s
        \bigl(\bx,\blmbd;\bz,\bmu,\by(\bx,\blmbd)\bigr)
        &=
        \nabla f(\bx)
        +J_{\calA}(\bx)^\top\bq(\bx,\blmbd)
        +\frac{1}{\gamma}(\bx-\bz), \\
        \nabla_{\blmbd}\mathcal P_s
        \bigl(\bx,\blmbd;\bz,\bmu,\by(\bx,\blmbd)\bigr)
        &=
        \calA(\bx)-\by(\bx,\blmbd)
        -\frac{1}{s}(\blmbd-\bmu),
    \end{aligned}
\end{equation} where $ \bq(\bx,\blmbd) := \blmbd+\rho\bigl[\calA(\bx)-\by(\bx,\blmbd)\bigr]$, whose  optimality condition gives
\begin{equation}
    \bq(\bx,\blmbd)
    \in\partial h\bigl(\by(\bx,\blmbd)\bigr).
    \label{eq:certificate-membership}
\end{equation}

Consequently, the first two updates of
Algorithm~\ref{alg:ddlalm} can be written explicitly as
\begin{equation}\label{eq:explicit-updates}
    \begin{aligned}
        \bx^{k+1}
        &=
        \bx^k-\eta_k
        \left[
            \nabla f(\bx^k)
            +J_{\calA}(\bx^k)^\top\bq(\bx^k,\blmbd^k)
            +\frac{1}{\gamma}(\bx^k-\bz^k)
        \right], \\
        \blmbd^{k+1}
        &=
        \blmbd^k+\tau_k
        \left[
            \calA(\bx^{k+1})
            -\by(\bx^{k+1},\blmbd^k)
            -\frac{1}{s_k}(\blmbd^k-\bmu^k)
        \right].
    \end{aligned}
\end{equation}

\begin{algorithm}[t]
    \caption{Dual-Damping Linearized Augmented Lagrangian Method (DDLALM)}
    \label{alg:ddlalm}
    \small
    \begin{algorithmic}[1]
        \STATE \textbf{Input:} Initial
        \(\bx^0,\bz^0\in\RR^n\) and
        \(\blmbd^0,\bmu^0\in\RR^m\); 
        Parameters: \(\rho>0\) and \(\gamma>0\);
        sequences: \(\{\eta_k\}\), \(\{\tau_k\}\), \(\{\alpha_k\}\), and
        \(\{\beta_k\}\);
        and iteration limit \(K\).
        \FOR{\(k=0,1,\ldots,K-1\)}
            \STATE  \(\bx^{k+1}=\bx^k- \eta_k \nabla_{\bx} \mathcal P_{s_k}(\bx^{k}, \blmbd^{k}; \bz^{k}, \bmu^{k}, \by(\bx^k, \blmbd^k))\);
            \STATE  \(\blmbd^{k+1}=\blmbd^k+\tau_k
            \nabla_{\blmbd}\mathcal P_{s_k}
            (\bx^{k+1},\blmbd^{k};\bz^{k},\bmu^{k},
            \by(\bx^{k+1},\blmbd^k))\).
            \STATE 
            \(\bz^{k+1}=\bz^k+ \alpha_k(\bx^{k+1}-\bz^k)\).
            \STATE  \(\bmu^{k+1}=\bmu^k+ \beta_k(\blmbd^{k+1}-\bmu^k)\).
        \ENDFOR
    \end{algorithmic}
\end{algorithm}

We use fixed primal and dual stepsizes together with diminishing center
stepsizes. Choose \(\rho>0\) and \(\gamma>0\) satisfying
Proposition~\ref{prop:curvature-smoothness}. With \(L_x\),
\(\mu_\gamma\), and \(L_{d,0}\) defined in
Section~\ref{sec:complexity-analysis}, one admissible choice is
\begin{equation}\label{eq:ddlalm-parameter-schedule}
\begin{aligned}
    s_k &= c_s(k+1)^{1/3},
    \qquad
    \eta = \frac{1}{2L_x},
    \qquad
    \tau =
    \left(
        2L_{d,0}
        +\frac{72L_xG_{\cal A}^2}{\mu_\gamma^2}
    \right)^{-1},
    \\
    \alpha_k &=
    \frac{\min\{s_0,\gamma\mu_\gamma\tau\}}{36s_k},
    \qquad
    \beta_k=\frac{\theta_\mu}{k+k_0},
\end{aligned}
\end{equation}
where \(c_s>0\), \(\theta_\mu\in(0,1]\), and \(k_0\ge1\) are constants
independent of \(K\). Consequently,
\[
    s_k=\Theta((k+1)^{1/3}),
    \qquad
    \alpha_k=\Theta((k+1)^{-1/3}),
    \qquad
    \beta_k=\Theta((k+1)^{-1}).
\]

\begin{theorem}
\label{thm:main}
Suppose that Assumptions~\ref{ass:level-boundedness},
\ref{ass:smoothness}, and~\ref{ass:h} hold.
Let the sequence generated by \Cref{alg:ddlalm} use
\eqref{eq:ddlalm-parameter-schedule}, with $\rho$ and $\gamma$ as specified
above. Then it holds that
\[
\max\left\{
\left[
    \left\|
        \nabla f({\bx}^{I_K+1})
        +
        J_{\cal A}({\bx}^{I_K+1})^\top{\bq}^{I_K}
    \right\|^2
\right],
\;
\left[
    \left\|
        \cal A({\bx}^{I_K+1})-{\by}^{I_K+1}
    \right\|^2
\right]
\right\}
=
\mathcal O(K^{-2/3}),
\]
with
\({\bq}^{I_K}\in\partial h({\by}^{I_K+1})\).
Consequently, given \(\epsilon>0\), the iteration complexity of
\Cref{alg:ddlalm} to obtain an \(\epsilon\)-KKT point
is of order \(\mathcal O(\epsilon^{-3})\).
\end{theorem}
