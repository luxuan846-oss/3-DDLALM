\section{Preliminaries}
% \subsection{Notation}

\textbf{Notation.} We use $\langle\cdot,\cdot\rangle$ and $\|\cdot\|$ for the Euclidean inner
product and norm.  For a matrix, $\|\cdot\|$ denotes the spectral norm.  For
$\calA=(A_1,\ldots,A_m)^\top$, let $J_{\calA}(\bx)\in\RR^{m\times n}$ denote its
Jacobian, whose $i$th row is $\nabla A_i(\bx)^\top$.  We write
$\sigma_{\min}(J_{\calA}(\bx))$ for the smallest singular value of
$J_{\calA}(\bx)$.  
% Throughout the paper, $\partial h$ denotes the limiting
% subdifferential of $h$.

For a proper closed convex function \(h\), its convex subdifferential and
proximal mapping are, respectively,
\[
    \partial h(y)
    :=\{v:\ h(z)\ge h(y)+\langle v,z-y\rangle\ \text{for all }z\},
\]
\begin{equation}
    \prox_{\gamma h}(u)
    :=\argmin_{y\in\R^m}
       \left\{h(y)+\frac{1}{2\gamma}\|y-u\|^2\right\},
    \qquad \gamma>0.
    \label{eq:prox-definition}
\end{equation}

{
Let define the splitting residual as $\br(\bx,\by):=\calA(\bx)-\by$ and 
the KKT condition of the split formulation of Problem~\eqref{prob:main} as
\begin{equation}
\begin{aligned}
    \nabla f(\bx)+J_{\calA}(\bx)^\top\bz&=\zero,\\
    \bz&\in\partial h(\by),\\
    \calA(\bx)-\by &=\zero.
\end{aligned}
\notag
\end{equation}
Given an $\varepsilon>$
\begin{definition}[\(\epsilon\)-KKT point]
\label{def:eps-kkt}
For \(\epsilon>0\), a triple \((\bx,\by,\bz)\) is called an
\(\epsilon\)-KKT point of~\eqref{eq:split-problem} if there exists a variable 
\(\bz\in\partial h(\by)\) such that the following hold:
\begin{equation}
    \begin{cases}
        \|\nabla f(\bx)+J_{\calA}(\bx)^\top\bz\| \le \epsilon, \\
        \|\calA(\bx)-\by\| \le \epsilon,
    \end{cases}
    \label{eq:eps-kkt-residual}
\end{equation}
\end{definition}
} which is closely related to the $\varepsilon$-GS stationary point defined in \cite{bohm2021variable}. 

\subsection{Assumptions}
\label{sec:assumptions}

\begin{assumption}[Level-bounded objective]
\label{ass:level-boundedness}
Assume that $F$ is level-bounded, that is, for every $\alpha\in\RR$, the set
\[
    \calS_{\alpha}:=\{\bx\in\RR^n:F(\bx)\le\alpha\}
\]
is empty or compact.
\end{assumption}

\begin{assumption}
\label{ass:smoothness}
\begin{enumerate}[label={[\roman*].}]
    \item The function $f:\RR^n\to\RR$ is continuously differentiable,
    and there exists $L_f, G_f \ge0$ such that
    \[
        \|\nabla f(\bx)-\nabla f(\bx')\|
        \le
        L_f\|\bx-\bx'\|,
        \qquad \forall \bx,\bx'\in \RR^n,
    \] and 
    \[
    \|\nabla f(\bx)\| \leq  G_f,
    \qquad \forall \bx\in\RR^n.
    \]

    \item The mapping $\calA:\RR^n\to\RR^m$ is continuously
    differentiable, and there exist $L_{\calA},G_{\calA}\ge0$ such that
    \[
        \|J_{\calA}(\bx)-J_{\calA}(\bx')\|
        \le
        L_{\calA}\|\bx-\bx'\|,
        \qquad \forall \bx,\bx'\in\RR^n,
    \]
    and
    \[
        \|J_{\calA}(\bx)\|\le G_{\calA},
        \qquad \forall \bx\in\RR^n.
    \]
\end{enumerate}
\end{assumption}

\begin{assumption}
\label{ass:h}
The function $h:\RR^m\to\RR$ is proper, closed, convex, and 
$L_h$-Lipschitz, i.e.,
\begin{equation}
    \|\bxi\|\le L_h,
    \qquad
    \forall \by\in\RR^m,
    \quad \forall \bxi\in\partial h(\by).
    \label{eq:bounded-h-subgradients}
\end{equation}
\end{assumption}
