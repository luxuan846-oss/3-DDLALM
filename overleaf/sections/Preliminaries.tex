\section{Preliminaries}
\label{sec:preliminaries}

We collect the problem setup and oracle model for
Problem~\eqref{prob:main}. We then introduce the regularized augmented
Lagrangian that underlies our algorithms. Following~\cite{rockafellar1970convex}, for a proper closed convex function
\(h\colon\RR^m\to\RR\cup\{+\infty\}\), its subdifferential, convex conjugate,
and proximal mapping are
\begin{gather*}
    \partial h(\by):=\{\bv:\ h(\bw)\ge h(\by)+\langle\bv,\bw-\by\rangle
    \ \text{for all }\bw\},
    \qquad
    h^*(\bq):=\sup_{\by\in\RR^m}\{\langle\bq,\by\rangle-h(\by)\},\\
    \prox_{t h}(\bu)
    :=\argmin_{\by\in\RR^m}
       \Bigl\{h(\by)+\frac{1}{2t}\|\by-\bu\|^2\Bigr\}, 
    \qquad t>0.
\end{gather*}

\subsection{Problem setup}
\label{sec:assumptions}
We impose the following standing assumptions on Problem~\eqref{prob:main}. 
\begin{assumption}
\label{ass:smoothness}
There exist constants \(L_f,G_f,L_{\calA},G_{\calA}\ge0\) such that, for
all \(\bx,\bx'\in\RR^n\),
\[
\begin{aligned}
    \|\nabla f(\bx)-\nabla f(\bx')\|
    &\le L_f\|\bx-\bx'\|,
    &\|\nabla f(\bx)\|&\le G_f,\\
    \|J_{\calA}(\bx)-J_{\calA}(\bx')\|
    &\le L_{\calA}\|\bx-\bx'\|,
    &\|J_{\calA}(\bx)\|&\le G_{\calA}.
\end{aligned}
\]
\end{assumption}

\begin{assumption}
\label{ass:h}
The function \(h\colon\RR^m\to\RR\) is convex and \(L_h\)-Lipschitz, in the sense that
\[
    |h(\bu)-h(\bv)|\le L_h\|\bu-\bv\|,
    \qquad \bu,\bv\in\RR^m.
\]
\end{assumption}

We measure approximate stationarity of Problem~\eqref{prob:main} through its
KKT residual.
\begin{definition}
\label{def:eps-kkt}
Given \(\epsilon>0\), a triple \((\bx,\by,\bq)\) with
\(\bq\in\partial h(\by)\) is an \(\epsilon\)-KKT point of
Problem~\eqref{prob:main} if
\begin{equation}
    \mathcal{R}_{\mathrm{KKT}}(\bx,\by,\bq)
    :=
    \max\Bigl\{
        \bigl\|\nabla f(\bx)+J_{\calA}(\bx)^\top\bq\bigr\|,\;
        \|\calA(\bx)-\by\|
    \Bigr\} \leq \epsilon .
    \label{eq:kkt-residual}
\end{equation}
\end{definition}
At \(\epsilon=0\), \eqref{eq:kkt-residual} reduces to \(\by=\calA(\bx)\) and
\(\zero\in\nabla f(\bx)+J_{\calA}(\bx)^\top\partial h(\calA(\bx))\), the
first-order KKT condition of Problem~\eqref{prob:main}~\cite{li2020understanding}.

For complexity analysis, we specify the problem class~\cite{carmon2020lower,nemirovski1983problem}
and the oracle model for Problem~\eqref{prob:main}.
\begin{definition}
\label{def:problem-class}
Let \(\Delta>0\) and
\(\mathbf C=(L_f,G_f,L_{\calA},G_{\calA},L_h)\in\RR^5_{\ge0}\).
We denote by \(\mathcal I(\mathbf C,\Delta)\) the set of all triples
\((f,\calA,h)\), over all dimensions \(n,m\in\NN\), such that
Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold with the constants
in \(\mathbf C\) and
\(
    F(\zero)-\inf_{\bx\in\RR^n}F(\bx)\le\Delta .
\)
\end{definition}

\begin{definition}
\label{def:oracle-model}
We assume access to a deterministic first-order oracle
\[
    \mathcal{O}_{f,\mathcal{A}}(\bx)
    :=
    \bigl(f(\bx),\nabla f(\bx),\mathcal{A}(\bx),J_{\mathcal{A}}(\bx)\bigr),
\]
and an exact proximal oracle
\[
    \mathcal{O}_{h}(\bv,t)
    :=
    \operatorname{prox}_{t h}(\bv),
    \qquad t>0.
\]
Each call to either oracle counts as one oracle evaluation.
\end{definition}

\subsection{\texorpdfstring{Regularized augmented Lagrangian for Problem~\eqref{prob:main}}{Regularized augmented Lagrangian for Problem (P)}}
\label{subsec:regularized-al}
For \(\rho>0\), the augmented Lagrangian of Problem~\eqref{prob:main} is
\(\calL_\rho(\bx,\by,\blmbd):=f(\bx)+h(\by)+\langle\blmbd,\calA(\bx)-\by\rangle+\frac{\rho}{2}\|\calA(\bx)-\by\|^2\).
Eliminating \(\by\) yields the reduced augmented
Lagrangian~\cite{rockafellar1976augmented}, 
\begin{equation}
    \calD_\rho(\bx,\blmbd)
    :=\min_{\by\in\RR^m}\calL_\rho(\bx,\by,\blmbd),
    \label{eq:reduced-al}
\end{equation}
with unique minimizer and associated multiplier
\begin{equation}
    \by(\bx,\blmbd):=\prox_{h/\rho}\Bigl(\calA(\bx)+\frac{\blmbd}{\rho}\Bigr),
    \qquad
    \bq(\bx,\blmbd):=\blmbd+\rho\bigl[\calA(\bx)-\by(\bx,\blmbd)\bigr].
    \label{eq:prox-multiplier}
\end{equation}
The next lemma records the smoothness of \(\calD_\rho\) and shows that
maximizing it over \(\blmbd\) recovers \(F\).

\begin{lemma}
\label{lem:reduced-al}
Suppose that Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold, and let
\(\rho>0\).
\begin{enumerate}[label=(\alph*)] 
\item \(\calD_\rho\) is continuously differentiable on \(\RR^n\times\RR^m\),
with \(\bq(\bx,\blmbd)\in\partial h(\by(\bx,\blmbd))\) and
\begin{equation}
    \nabla_{\bx}\calD_\rho(\bx,\blmbd)
    =\nabla f(\bx)+J_{\calA}(\bx)^\top\bq(\bx,\blmbd),
    \quad
    \nabla_{\blmbd}\calD_\rho(\bx,\blmbd)
    =\calA(\bx)-\by(\bx,\blmbd)
    =\frac{\bq(\bx,\blmbd)-\blmbd}{\rho}.
    \label{eq:reduced-al-gradient}
\end{equation}
For each \(\bx\in\RR^n\), \(\calD_\rho(\bx,\cdot)\) is concave and
\(\nabla_{\blmbd}\calD_\rho(\bx,\cdot)\) is \((1/\rho)\)-Lipschitz
continuous.
\item For every \((\bx,\blmbd)\in\RR^n\times\RR^m\),
\begin{equation}
    \calD_\rho(\bx,\blmbd)
    =\max_{\bq\in\RR^m}\Bigl\{\calL(\bx,\bq)-\frac{1}{2\rho}\|\bq-\blmbd\|^2\Bigr\},
    \label{eq:dual-smoothing}
\end{equation}
attained uniquely at \(\bq(\bx,\blmbd)\). Consequently,
\(\max_{\blmbd\in\RR^m}\calD_\rho(\bx,\blmbd)=F(\bx)\) for all \(\bx\in\RR^n\),
with maximizer set \(\partial h(\calA(\bx))\).
\end{enumerate}
\end{lemma}

By Lemma~\ref{lem:reduced-al}, Problem~\eqref{prob:main} reduces to the
unconstrained smooth NC-C minimax problem \(\min_{\bx\in\RR^n}\max_{\blmbd\in\RR^m}\calD_\rho(\bx,\blmbd)\).
By~\eqref{eq:kkt-residual} and~\eqref{eq:reduced-al-gradient},
\begin{equation}
    \mathcal R_{\mathrm{KKT}}\bigl(\bx,\by(\bx,\blmbd),\bq(\bx,\blmbd)\bigr)
    =\max\bigl\{
        \|\nabla_{\bx}\calD_\rho(\bx,\blmbd)\|,\;
        \|\nabla_{\blmbd}\calD_\rho(\bx,\blmbd)\|
    \bigr\},
    \label{eq:kkt-gs-identity}
\end{equation}
so any \(\epsilon\)-GS point
\((\bx,\blmbd)\)~\cite{pan2026optimal}, where the right-hand
side is at most \(\epsilon\), directly gives an \(\epsilon\)-KKT point
\((\bx,\by(\bx,\blmbd),\bq(\bx,\blmbd))\) of Problem~\eqref{prob:main}. 

In this view, a gradient-type linearized
ALM replaces the
exact \(\bx\)-minimization of classical
ALM~\cite{hestenes1969multiplier,powell1969method,rockafellar1973multiplier}
by one gradient step and becomes alternating GDA on
\(\calD_\rho\),
\begin{equation}
\left\{
\begin{aligned} 
    \bx^{k+1}
    &=\bx^k-\eta\nabla_{\bx}\calD_\rho(\bx^k,\blmbd^k),\\
    \blmbd^{k+1}
    &=\blmbd^k+\rho\nabla_{\blmbd}\calD_\rho(\bx^{k+1},\blmbd^k)
    =\bq(\bx^{k+1},\blmbd^k),
\end{aligned} 
\right.
\label{eq:alm-reduced}
\end{equation} 
with \(\eta>0\). Since single-timescale GDA can fail to
converge even on bilinear problems~\cite{daskalakis2018training},
we adopt the primal~\cite{zhang2020smoothed} and
dual~\cite{lu2020hybrid,xu2023unified,zheng2023universal,li2026smoothing}
quadratic regularizers of smooth minimax methods and define the
regularized augmented Lagrangian for Problem~\eqref{prob:main},
\begin{equation}
    \mathcal P_s(\bx,\blmbd;\bz,\bmu)
    :=\calD_\rho(\bx,\blmbd)+\frac{1}{2\gamma}\|\bx-\bz\|^2
    -\frac{1}{2s}\|\blmbd-\bmu\|^2,
    \label{eq:potential}
\end{equation}
where \(\gamma,s>0\) and \((\bz,\bmu)\in\RR^n\times\RR^m\). Set \(L_{\calD,\bx}:=L_f+L_{\calA}L_h+\rho G_{\calA}^{2}\),
\(\mu_\gamma:=\gamma^{-1}-L_{\calD,\bx}\),
\(L_x:=\gamma^{-1}+L_{\calD,\bx}\), and \(L_\lambda:=\rho^{-1}+s^{-1}\).
The next proposition records the curvature, block-Lipschitz smoothness, and
saddle-point properties of \(\mathcal P_s\), which follow directly from
standard convex analysis~\cite{rockafellar1970convex}.
\begin{proposition}
\label{prop:curvature-smoothness}
Suppose that Assumptions~\ref{ass:smoothness} and~\ref{ass:h} hold, and let
\(\gamma>0\) satisfy \(\gamma L_{\calD,\bx}<1/2\). Then, for every \(s>0\) and
\((\bz,\bmu)\in\RR^n\times\RR^m\), the following statements hold.
\begin{enumerate}[label=(\alph*)]
\item \(\mathcal P_s(\cdot,\cdot;\bz,\bmu)\) is \(\mu_\gamma\)-strongly
convex in \(\bx\) and \(s^{-1}\)-strongly concave in \(\blmbd\). Moreover, for
all \(\bx,\bx'\in\RR^n\) and \(\blmbd,\blmbd'\in\RR^m\),
\begin{align*}
    \bigl\|
    \nabla_{\bx}\mathcal{P}_s(\bx,\blmbd;\bz,\bmu)
    -\nabla_{\bx}\mathcal{P}_s(\bx',\blmbd';\bz,\bmu)
    \bigr\|
    &\le L_x\|\bx-\bx'\|+G_{\calA}\|\blmbd-\blmbd'\|,\\
    \bigl\|
    \nabla_{\blmbd}\mathcal{P}_s(\bx,\blmbd;\bz,\bmu)
    -\nabla_{\blmbd}\mathcal{P}_s(\bx',\blmbd';\bz,\bmu)
    \bigr\|
    &\le G_{\calA}\|\bx-\bx'\|+L_\lambda\|\blmbd-\blmbd'\|.
\end{align*}
In particular, \(\mathcal P_s(\cdot,\cdot;\bz,\bmu)\) is \(L_x\)-smooth in
\(\bx\) and \(L_\lambda\)-smooth in \(\blmbd\).
\item \(\mathcal P_s(\cdot,\cdot;\bz,\bmu)\) admits a unique saddle point
\((\bx^\star,\blmbd^\star)\in\RR^n\times\RR^m\) and satisfies
\[
    \min_{\bx\in\RR^n}\max_{\blmbd\in\RR^m}\mathcal P_s(\bx,\blmbd;\bz,\bmu)
    =\max_{\blmbd\in\RR^m}\min_{\bx\in\RR^n}\mathcal P_s(\bx,\blmbd;\bz,\bmu)
    =\mathcal P_s(\bx^\star,\blmbd^\star;\bz,\bmu).
\]
\end{enumerate}
\end{proposition}
